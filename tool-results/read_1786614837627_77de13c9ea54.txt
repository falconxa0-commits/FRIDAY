   149→   149→    async def start(self) -> RecoveryReport:
   150→   150→        """Initialize: load persistent state, prepare subsystems."""
   151→   151→        if self._started:
   152→   152→            return self._recovery_report
   153→   153→        async with self._lock:
   154→   154→            self._started = True
   155→   155→            # Load from persistence
   156→   156→            items, report = await self._persistence.load()
   157→   157→            self._recovery_report = report
   158→   158→            for mem, prov in items:
   159→   159→                # Re-verify integrity on load
   160→   160→                if not self._immune.verify_integrity(mem, prov):
   161→   161→                    report.quarantined_count += 1
   162→   162→                    report.quarantined_ids.append(mem.id.value)
   163→   163→                    # Quarantine in memory (state mutation)
   164→   164→                    try:
   165→   165→                        validate_transition(mem.state, MemoryState.QUARANTINED)
   166→   166→                        mem.state = MemoryState.QUARANTINED
   167→   167→                    except LifecycleError:
   168→   168→                        pass
   169→   169→                tenant = self._get_or_create_tenant(mem.metadata.tenant_id)
   170→   170→                tenant.memories[mem.id.value] = (mem, prov)
   171→   171→            logger.info(
   172→   172→                "LivingMemoryManager started — loaded %d, quarantined %d",
   173→   173→                report.loaded_count, report.quarantined_count,
   174→   174→            )
   175→   175→        # Emit start event
   176→   176→        await self._emit_event("memory.system.started", "", "", "default", {
   177→   177→            "loaded_count": report.loaded_count,
   178→   178→            "quarantined_count": report.quarantined_count,
   179→   179→        })
   180→   180→        return report
   181→   181→
   182→   182→    async def stop(self) -> None:
   183→   183→        """Persist all state and shutdown."""
   184→   184→        if not self._started:
   185→   185→            return
   186→   186→        async with self._lock:
   187→   187→            # Persist all memories (re-seal integrity hash first)
   188→   188→            for tenant in self._tenants.values():
   189→   189→                for mid, (mem, prov) in tenant.memories.items():
   190→   190→                    try:
   191→   191→                        await self._persist(mem, prov)
   192→   192→                    except Exception as e:
   193→   193→                        logger.error("Failed to persist %s: %s", mid[:8], e)
   194→   194→            self._started = False
   195→   195→        await self._emit_event("memory.system.stopped", "", "", "default", {})
   196→   196→        logger.info("LivingMemoryManager stopped")

   214→   214→    async def remember(
   215→   215→        self,
   216→   216→        memory: Memory,
   217→   217→        ctx: AuthorizationContext,
   218→   218→        provenance: Optional[Provenance] = None,
   219→   219→        idempotency_key: str = "",
   220→   220→        source: str = "",
   221→   221→        notes: str = "",
   222→   222→    ) -> Memory:
   223→   223→        """Create + persist a memory. Validates via immune system.
   224→   224→
   225→   225→        Args:
   226→   226→            memory: The memory to store (state must be CREATED).
   227→   227→            ctx: Authorization context for the caller.
   228→   228→            provenance: Optional pre-built provenance chain. If omitted, a
   229→   229→                fresh chain with a single "create" entry is created.
   230→   230→            idempotency_key: Optional replay-protection key.
   231→   231→            source: Source description for provenance.
   232→   232→            notes: Free-form notes for provenance.
   233→   233→
   234→   234→        Returns:
   235→   235→            The stored memory (with integrity_hash set).
   236→   236→
   237→   237→        Raises:
   238→   238→            AuthorizationError: caller not authorized.
   239→   239→            ImmuneRejection: memory failed validation.
   240→   240→            ContradictionError: a contradiction was detected (not silently overwritten).
   241→   241→            ResourceLimitError: tenant at memory cap.
   242→   242→        """
   243→   243→        if not self._started:
   244→   244→            raise RuntimeError("Manager not started; call start() first")
   245→   245→
   246→   246→        # Authorize write
   247→   247→        self._authz.authorize_write(ctx, memory.metadata.tenant_id)
   248→   248→
   249→   249→        # Build provenance if missing
   250→   250→        if provenance is None:
   251→   251→            provenance = Provenance()
   252→   252→            provenance.append(
   253→   253→                actor_id=ctx.citizen_id,
   254→   254→                action="create",
   255→   255→                source=source,
   256→   256→                process="living_memory",
   257→   257→                notes=notes,
   258→   258→                correlation_id=memory.metadata.correlation_id or new_correlation_id(),
   259→   259→            )
   260→   260→        elif provenance.is_empty:
   261→   261→            provenance.append(
   262→   262→                actor_id=ctx.citizen_id, action="create",
   263→   263→                source=source, process="living_memory",
   264→   264→                correlation_id=memory.metadata.correlation_id or new_correlation_id(),
   265→   265→            )
   266→   266→
   267→   267→        # Get tenant state (create if missing)
   268→   268→        tenant = self._get_or_create_tenant(memory.metadata.tenant_id)
   269→   269→
   270→   270→        # Check tenant memory cap (homeostasis)
   271→   271→        async with self._lock:
   272→   272→            if len(tenant.memories) >= self._config.max_memories_per_tenant:
   273→   273→                raise ResourceLimitError(
   274→   274→                    f"Tenant {memory.metadata.tenant_id} at memory cap "
   275→   275→                    f"{self._config.max_memories_per_tenant}"
   276→   276→                )
   277→   277→            existing_ids = set(tenant.memories.keys())
   278→   278→
   279→   279→        # Run immune system scan
   280→   280→        report = self._immune.scan(
   281→   281→            memory=memory,
   282→   282→            provenance=provenance,
   283→   283→            existing_ids=existing_ids,
   284→   284→            seen_idempotency_keys=self._seen_idempotency,
   285→   285→            idempotency_key=idempotency_key,
   286→   286→            expected_tenant_id=ctx.tenant_id,
   287→   287→        )
   288→   288→        if not report.accepted:
   289→   289→            # Quarantine the memory (don't silently drop)
   290→   290→            if self._config.quarantine_enabled:
   291→   291→                try:
   292→   292→                    validate_transition(memory.state, MemoryState.QUARANTINED)
   293→   293→                    memory.state = MemoryState.QUARANTINED
   294→   294→                    tenant.memories[memory.id.value] = (memory, provenance)
   295→   295→                    await self._emit_event(
   296→   296→                        "memory.quarantined", memory.id.value,
   297→   297→                        memory.type.value, memory.metadata.tenant_id,
   298→   298→                        {"reason": report.reason, "actor_id": ctx.citizen_id},
   299→   299→                    )
   300→   300→                except LifecycleError:
   301→   301→                    pass
   302→   302→            self._observability.metrics.inc("memory.rejections", reason=report.reason)
   303→   303→            raise ImmuneRejection(report.reason, report.to_dict())
   304→   304→
   305→   305→        # Track idempotency key
   306→   306→        if idempotency_key:
   307→   307→            self._seen_idempotency.add(idempotency_key)
   308→   308→            # Bound the cache size
   309→   309→            if len(self._seen_idempotency) > self._max_idempotency_cache:
   310→   310→                # Drop oldest 20%
   311→   311→                self._seen_idempotency = set(list(self._seen_idempotency)[int(self._max_idempotency_cache * 0.2):])
   312→   312→
   313→   313→        # Detect contradictions (semantic + procedural)
   314→   314→        if memory.type in (MemoryType.SEMANTIC, MemoryType.PROCEDURAL):
   315→   315→            candidates = self._build_contradiction_candidates(tenant, memory)
   316→   316→            contradictions = self._contradiction.detect(
   317→   317→                new_memory_id=memory.id.value,
   318→   318→                new_payload=memory.payload,
   319→   319→                new_type=memory.type.value,
   320→   320→                candidates=candidates,
   321→   321→            )
   322→   322→            for c in contradictions:
   323→   323→                if c.conflict_type.value == "duplicate":
   324→   324→                    # Duplicate is not a hard error; record + skip
   325→   325→                    self._observability.metrics.inc("memory.duplicates")
   326→   326→                    continue
   327→   327→                # Real conflict — store + raise
   328→   328→                tenant.contradictions[c.id] = c
   329→   329→                await self._emit_event(
   330→   330→                    "memory.contradiction_detected", memory.id.value,
   331→   331→                    memory.type.value, memory.metadata.tenant_id,
   332→   332→                    {
   333→   333→                        "contradiction_id": c.id,
   334→   334→                        "conflict_type": c.conflict_type.value,
   335→   335→                        "existing_memory_id": c.memory_a_id,
   336→   336→                        "description": c.description,
   337→   337→                        "actor_id": ctx.citizen_id,
   338→   338→                    },
   339→   339→                )
   340→   340→                self._observability.metrics.inc("memory.contradictions_detected")
   341→   341→                if len(tenant.contradictions) > self._config.max_contradictions_per_tenant:
   342→   342→                    raise ResourceLimitError("Too many unresolved contradictions")
   343→   343→                raise ContradictionError(c.description)
   344→   344→
   345→   345→        # Transition CREATED → ACTIVE
   346→   346→        validate_transition(memory.state, MemoryState.ACTIVE)
   347→   347→        memory.state = MemoryState.ACTIVE
   348→   348→        memory.metadata.updated_at = now_utc()
   349→   349→
   350→   350→        # Persist (re-seals integrity hash)
   351→   351→        await self._persist(memory, provenance)
   352→   352→
   353→   353→        async with self._lock:
   354→   354→            tenant.memories[memory.id.value] = (memory, provenance)
   355→   355→            tenant.write_count += 1
   356→   356→
   357→   357→        # Update metrics + emit event
   358→   358→        self._observability.metrics.inc("memory.writes", type=memory.type.value)
   359→   359→        self._observability.metrics.inc_gauge(
   360→   360→            "memory.active_count",
   361→   361→            labels={"type": memory.type.value, "state": memory.state.value},
   362→   362→        )
   363→   363→        await self._emit_event(
   364→   364→            "memory.created", memory.id.value,
   365→   365→            memory.type.value, memory.metadata.tenant_id,
   366→   366→            {"actor_id": ctx.citizen_id, "state": memory.state.value},
   367→   367→            correlation_id=memory.metadata.correlation_id,
   368→   368→        )
   369→   369→
   370→   370→        # Optional auto-decay
   371→   371→        if self._config.auto_decay_on_write and \
   372→   372→           tenant.write_count % self._config.decay_interval_writes == 0:
   373→   373→            asyncio.create_task(self.decay(ctx))
   374→   374→
   375→   375→        return memory
   376→   376→
   377→   377→    async def recall(
   378→   378→        self,
   379→   379→        memory_id: str,
   380→   380→        ctx: AuthorizationContext,
   381→   381→    ) -> Optional[Memory]:
   382→   382→        """Retrieve a memory by ID. Touches access stats."""
   383→   383→        if not self._started:
   384→   384→            raise RuntimeError("Manager not started")
   385→   385→        # Validate ID format (immune check)
   386→   386→        try:
   387→   387→            MemoryID.from_string(memory_id)
   388→   388→        except ValidationError:
   389→   389→            return None
   390→   390→
   391→   391→        tenant = self._get_or_create_tenant(ctx.tenant_id)
   392→   392→        async with self._lock:
   393→   393→            entry = tenant.memories.get(memory_id)
   394→   394→            if not entry:
   395→   395→                return None
   396→   396→            mem, prov = entry
   397→   397→            # Authorize read
   398→   398→            self._authz.authorize_read(ctx, mem.metadata.tenant_id, mem.metadata.owner_id)
   399→   399→            # Skip terminal/inaccessible states
   400→   400→            if not MemoryState.is_accessible(mem.state):
   401→   401→                return None
   402→   402→            mem.metadata.touch()
   403→   403→            self._observability.metrics.inc("memory.reads", type=mem.type.value)
   404→   404→            await self._emit_event(
   405→   405→                "memory.recalled", mem.id.value,
   406→   406→                mem.type.value, mem.metadata.tenant_id,
   407→   407→                {"actor_id": ctx.citizen_id},
   408→   408→            )
   409→   409→            return mem

   411→   411→    async def recall_by_type(
   412→   412→        self,
   413→   413→        memory_type: MemoryType,
   414→   414→        ctx: AuthorizationContext,
   415→   415→        limit: int = 100,
   416→   416→        state_filter: Optional[MemoryState] = None,
   417→   417→    ) -> List[Memory]:
   418→   418→        """List memories of a given type."""
   419→   419→        if limit < 1 or limit > 1000:
   420→   420→            raise ValidationError("limit must be 1..1000")
   421→   421→        tenant = self._get_or_create_tenant(ctx.tenant_id)
   422→   422→        async with self._lock:
   423→   423→            results: List[Memory] = []
   424→   424→            for mem, _ in tenant.memories.values():
   425→   425→                if mem.type != memory_type:
   426→   426→                    continue
   427→   427→                if state_filter and mem.state != state_filter:
   428→   428→                    continue
   429→   429→                if not MemoryState.is_accessible(mem.state):
   430→   430→                    continue
   431→   431→                # Authorize read
   432→   432→                try:
   433→   433→                    self._authz.authorize_read(ctx, mem.metadata.tenant_id, mem.metadata.owner_id)
   434→   434→                except AuthorizationError:
   435→   435→                    continue
   436→   436→                results.append(mem)
   437→   437→                if len(results) >= limit:
   438→   438→                    break
   439→   439→            return results

   441→   441→    async def search(
   442→   442→        self,
   443→   443→        ctx: AuthorizationContext,
   444→   444→        memory_type: Optional[MemoryType] = None,
   445→   445→        tags: Optional[List[str]] = None,
   446→   446→        min_confidence: float = 0.0,
   447→   447→        min_importance: float = 0.0,
   448→   448→        limit: int = 50,
   449→   449→    ) -> List[Memory]:
   450→   450→        """Filtered retrieval (no vector search — pure metadata filter)."""
   451→   451→        if limit < 1 or limit > 1000:
   452→   452→            raise ValidationError("limit must be 1..1000")
   453→   453→        tenant = self._get_or_create_tenant(ctx.tenant_id)
   454→   454→        async with self._lock:
   455→   455→            results: List[Memory] = []
   456→   456→            for mem, _ in tenant.memories.values():
   457→   457→                if memory_type and mem.type != memory_type:
   458→   458→                    continue
   459→   459→                if not MemoryState.is_accessible(mem.state):
   460→   460→                    continue
   461→   461→                if mem.metadata.confidence < min_confidence:
   462→   462→                    continue
   463→   463→                if mem.metadata.importance < min_importance:
   464→   464→                    continue
   465→   465→                if tags:
   466→   466→                    md_tags = set(mem.metadata.tags)
   467→   467→                    if not set(tags).issubset(md_tags):
   468→   468→                        continue
   469→   469→                try:
   470→   470→                    self._authz.authorize_read(ctx, mem.metadata.tenant_id, mem.metadata.owner_id)
   471→   471→                except AuthorizationError:
   472→   472→                    continue
   473→   473→                results.append(mem)
   474→   474→            # Sort by importance desc, then confidence desc
   475→   475→            results.sort(
   476→   476→                key=lambda m: (-m.metadata.importance, -m.metadata.confidence)
   477→   477→            )
   478→   478→            return results[:limit]

   480→   480→    async def reinforce(
   481→   481→        self,
   482→   482→        memory_id: str,
   483→   483→        ctx: AuthorizationContext,
   484→   484→        delta: float = 0.1,
   485→   485→        notes: str = "",
   486→   486→    ) -> Optional[Memory]:
   487→   487→        """Reinforce a memory: bump confidence + importance."""
   488→   488→        if not 0.0 <= delta <= 1.0:
   489→   489→            raise ValidationError("delta must be in [0, 1]")
   490→   490→        tenant = self._get_or_create_tenant(ctx.tenant_id)
   491→   491→        async with self._lock:
   492→   492→            entry = tenant.memories.get(memory_id)
   493→   493→            if not entry:
   494→   494→                return None
   495→   495→            mem, prov = entry
   496→   496→            self._authz.authorize_read(ctx, mem.metadata.tenant_id, mem.metadata.owner_id)
   497→   497→            self._authz.authorize_capability(
   498→   498→                ctx, MemoryCapability.MEMORY_REINFORCE, mem.metadata.tenant_id
   499→   499→            )
   500→   500→            # State must be accessible
   501→   501→            if not MemoryState.is_accessible(mem.state):
   502→   502→                raise LifecycleError(
   503→   503→                    f"Cannot reinforce memory in state {mem.state.value}"
   504→   504→                )
   505→   505→            # Apply reinforcement
   506→   506→            old_state = mem.state
   507→   507→            mem.metadata.reinforce(delta)
   508→   508→            # Transition to REINFORCED (allowed from ACTIVE/REINFORCED/CONSOLIDATED/DECAYING)
   509→   509→            try:
   510→   510→                validate_transition(mem.state, MemoryState.REINFORCED)
   511→   511→                mem.state = MemoryState.REINFORCED
   512→   512→            except LifecycleError:
   513→   513→                # Stay in current state if transition not allowed
   514→   514→                pass
   515→   515→            prov.append(
   516→   516→                actor_id=ctx.citizen_id, action="reinforce",
   517→   517→                process="living_memory", notes=notes or f"reinforce +{delta}",
   518→   518→                correlation_id=mem.metadata.correlation_id,
   519→   519→            )
   520→   520→            # Persist
   521→   521→            await self._persist(mem, prov)
   522→   522→            self._observability.metrics.inc("memory.reinforcements", type=mem.type.value)
   523→   523→            await self._emit_event(
   524→   524→                "memory.reinforced", mem.id.value,
   525→   525→                mem.type.value, mem.metadata.tenant_id,
   526→   526→                {"actor_id": ctx.citizen_id, "delta": delta,
   527→   527→                 "old_state": old_state.value, "new_state": mem.state.value},
   528→   528→            )
   529→   529→            return mem

   535→   535→    async def consolidate(
   536→   536→        self,
   537→   537→        ctx: AuthorizationContext,
   538→   538→        memory_ids: Optional[List[str]] = None,
   539→   539→    ) -> ConsolidationResult:
   540→   540→        """Run a consolidation pass.
   541→   541→
   542→   542→        If memory_ids is None, scans all memories in the caller's tenant.
   543→   543→        """
   544→   544→        self._authz.authorize_capability(ctx, MemoryCapability.MEMORY_CONSOLIDATE)
   545→   545→        tenant = self._get_or_create_tenant(ctx.tenant_id)
   546→   546→        async with self._lock:
   547→   547→            memories = []
   548→   548→            for mid, (mem, _) in tenant.memories.items():
   549→   549→                if memory_ids and mid not in memory_ids:
   550→   550→                    continue
   551→   551→                if not MemoryState.is_accessible(mem.state):
   552→   552→                    continue
   553→   553→                memories.append(mem)
   554→   554→        # Score candidates
   555→   555→        result = self._consolidation.score_candidates(memories)
   556→   556→        # Apply transitions for eligible candidates
   557→   557→        for cand in result.candidates:
   558→   558→            if not cand.eligible:
   559→   559→                continue
   560→   560→            if not self._consolidation.eligible_for_transition(cand):
   561→   561→                continue
   562→   562→            async with self._lock:
   563→   563→                entry = tenant.memories.get(cand.memory_id)
   564→   564→                if not entry:
   565→   565→                    continue
   566→   566→                mem, prov = entry
   567→   567→                # Transition ACTIVE/REINFORCED/DECAYING → CONSOLIDATING → CONSOLIDATED
   568→   568→                try:
   569→   569→                    validate_transition(mem.state, MemoryState.CONSOLIDATING)
   570→   570→                    mem.state = MemoryState.CONSOLIDATING
   571→   571→                    validate_transition(mem.state, MemoryState.CONSOLIDATED)
   572→   572→                    mem.state = MemoryState.CONSOLIDATED
   573→   573→                    mem.metadata.updated_at = now_utc()
   574→   574→                    prov.append(
   575→   575→                        actor_id=ctx.citizen_id, action="consolidate",
   576→   576→                        process="consolidation_engine",
   577→   577→                        notes=f"score={cand.score:.3f}",
   578→   578→                        correlation_id=mem.metadata.correlation_id,
   579→   579→                    )
   580→   580→                    await self._persist(mem, prov)
   581→   581→                    self._observability.metrics.inc("memory.consolidations", type=mem.type.value)
   582→   582→                    await self._emit_event(
   583→   583→                        "memory.consolidated", mem.id.value,
   584→   584→                        mem.type.value, mem.metadata.tenant_id,
   585→   585→                        {"actor_id": ctx.citizen_id, "score": cand.score},
   586→   586→                    )
   587→   587→                except LifecycleError as e:
   588→   588→                    result.errors.append(f"{cand.memory_id}: {e}")
   589→   589→        return result

   591→   591→    async def decay(self, ctx: AuthorizationContext) -> DecayResult:
   592→   592→        """Run a decay pass."""
   593→   593→        self._authz.authorize_capability(ctx, MemoryCapability.MEMORY_ARCHIVE)
   594→   594→        tenant = self._get_or_create_tenant(ctx.tenant_id)
   595→   595→        async with self._lock:
   596→   596→            memories = [mem for mem, _ in tenant.memories.values()]
   597→   597→        # Score
   598→   598→        result = self._decay.score_candidates(memories)
   599→   599→        # Apply recommendations
   600→   600→        for cand in result.candidates:
   601→   601→            if cand.recommendation == "archive":
   602→   602→                async with self._lock:
   603→   603→                    entry = tenant.memories.get(cand.memory_id)
   604→   604→                    if not entry:
   605→   605→                        continue
   606→   606→                    mem, prov = entry
   607→   607→                    try:
   608→   608→                        validate_transition(mem.state, MemoryState.ARCHIVED)
   609→   609→                        mem.state = MemoryState.ARCHIVED
   610→   610→                        mem.metadata.updated_at = now_utc()
   611→   611→                        prov.append(
   612→   612→                            actor_id=ctx.citizen_id, action="archive",
   613→   613→                            process="decay_engine",
   614→   614→                            notes=f"decay_score={cand.decay_score:.3f}",
   615→   615→                            correlation_id=mem.metadata.correlation_id,
   616→   616→                        )
   617→   617→                        await self._persist(mem, prov)
   618→   618→                        self._observability.metrics.inc("memory.decays", action="archive")
   619→   619→                        await self._emit_event(
   620→   620→                            "memory.archived", mem.id.value,
   621→   621→                            mem.type.value, mem.metadata.tenant_id,
   622→   622→                            {"actor_id": ctx.citizen_id, "decay_score": cand.decay_score},
   623→   623→                        )
   624→   624→                    except LifecycleError as e:
   625→   625→                        result.errors.append(f"{cand.memory_id}: {e}")
   626→   626→            elif cand.recommendation == "forget":
   627→   627→                # Forget requires approval — defer to manager.forget()
   628→   628→                # Mark for review but don't auto-forget
   629→   629→                self._observability.metrics.inc("memory.decay_forget_pending")
   630→   630→        # Forced archival if over cap
   631→   631→        total_count = len(tenant.memories)
   632→   632→        if total_count > self._decay.config.max_total_memories:
   633→   633→            excess = total_count - self._decay.config.max_total_memories
   634→   634→            forced_ids = self._decay.select_forced_archive(memories, excess)
   635→   635→            for fid in forced_ids:
   636→   636→                async with self._lock:
   637→   637→                    entry = tenant.memories.get(fid)
   638→   638→                    if not entry:
   639→   639→                        continue
   640→   640→                    mem, prov = entry
   641→   641→                    try:
   642→   642→                        validate_transition(mem.state, MemoryState.ARCHIVED)
   643→   643→                        mem.state = MemoryState.ARCHIVED
   644→   644→                        mem.metadata.updated_at = now_utc()
   645→   645→                        prov.append(
   646→   646→                            actor_id=ctx.citizen_id, action="archive",
   647→   647→                            process="decay_engine_forced",
   648→   648→                            notes="forced archival due to capacity",
   649→   649→                            correlation_id=mem.metadata.correlation_id,
   650→   650→                        )
   651→   651→                        await self._persist(mem, prov)
   652→   652→                        result.forced_archive_count += 1
   653→   653→                    except LifecycleError:
   654→   654→                        pass
   655→   655→        tenant.last_decay_at = now_utc()
   656→   656→        return result

   662→   662→    async def forget(
   663→   663→        self,
   664→   664→        memory_id: str,
   665→   665→        ctx: AuthorizationContext,
   666→   666→        approval_id: Optional[str] = None,
   667→   667→        notes: str = "",
   668→   668→    ) -> bool:
   669→   669→        """Forget a memory. Requires Founder approval (Constitution Article 6).
   670→   670→
   671→   671→        If approval_id is provided, it must reference a PENDING/approved
   672→   672→        ApprovalRequest in the ApprovalGate (if available). If the approval

   690→   690→                if not req:
   691→   691→                    raise AuthorizationError(f"Approval {approval_id[:8]} not found")
   692→   692→                from .governance_compat import is_approval_approved  # local helper
   693→   693→                if not is_approval_approved(req):
   694→   694→                    raise AuthorizationError(
   695→   695→                        f"Approval {approval_id[:8]} not approved "
   696→   696→                        f"(status={req.status.value})"
   697→   697→                    )
   698→   698→            elif not ctx.is_founder:
   699→   699→                raise AuthorizationError(
   700→   700→                    "Forget without approval gate requires Founder rank"
   701→   701→                )
   702→   702→
   703→   703→            # Transition to FORGOTTEN
   704→   704→            try:
   705→   705→                validate_transition(mem.state, MemoryState.FORGOTTEN)
   706→   706→                mem.state = MemoryState.FORGOTTEN
   707→   707→                mem.metadata.updated_at = now_utc()
   708→   708→                prov.append(
   709→   709→                    actor_id=ctx.citizen_id, action="forget",
   710→   710→                    process="living_memory", notes=notes,
   711→   711→                    correlation_id=mem.metadata.correlation_id,
   712→   712→                )
   713→   713→                await self._persist(mem, prov)
   714→   714→                # Remove from association graph
   715→   715→                graph = tenant.association_graph
   716→   716→                if graph is not None:
   717→   717→                    await graph.remove_all_for(memory_id)
   718→   718→                self._observability.metrics.inc("memory.forgets")
   719→   719→                await self._emit_event(
   720→   720→                    "memory.forgotten", mem.id.value,
   721→   721→                    mem.type.value, mem.metadata.tenant_id,
   722→   722→                    {"actor_id": ctx.citizen_id, "approval_id": approval_id or ""},
   723→   723→                )
   724→   724→                return True
   725→   725→            except LifecycleError as e:
   726→   726→                logger.error("Forget failed: %s", e)
   727→   727→                return False

   805→   805→    async def resolve_contradiction(
   806→   806→        self,
   807→   807→        contradiction_id: str,
   808→   808→        ctx: AuthorizationContext,
   809→   809→        resolution: ContradictionStatus,
   810→   810→        approval_id: Optional[str] = None,
   811→   811→        notes: str = "",
   812→   812→    ) -> bool:
   813→   813→        """Resolve a contradiction. Requires Founder approval (Article 6)."""
   814→   814→        self._authz.authorize_capability(
   815→   815→            ctx, MemoryCapability.MEMORY_RESOLVE_CONTRADICTION
   816→   816→        )
   817→   817→        if resolution not in (
   818→   818→            ContradictionStatus.RESOLVED_KEEP_OLD,
   819→   819→            ContradictionStatus.RESOLVED_KEEP_NEW,
   820→   820→            ContradictionStatus.RESOLVED_MERGE,
   821→   821→            ContradictionStatus.RESOLVED_INVALID,
   822→   822→        ):
   823→   823→            raise ValidationError(f"Invalid resolution status: {resolution}")
   824→   824→
   825→   825→        # Check approval (if gate configured)
   826→   826→        if self._approval_gate is not None:
   827→   827→            if not approval_id:
   828→   828→                raise AuthorizationError("Resolve requires explicit approval_id")
   829→   829→            req = self._approval_gate.get_request(approval_id)
   830→   830→            if not req:
   831→   831→                raise AuthorizationError(f"Approval {approval_id[:8]} not found")
   832→   832→            from .governance_compat import is_approval_approved
   833→   833→            if not is_approval_approved(req):
   834→   834→                raise AuthorizationError(
   835→   835→                    f"Approval {approval_id[:8]} not approved"
   836→   836→                )
   837→   837→        elif not ctx.is_founder:
   838→   838→            raise AuthorizationError(
   839→   839→                "Resolve without approval gate requires Founder rank"
   840→   840→            )
   841→   841→
   842→   842→        tenant = self._get_or_create_tenant(ctx.tenant_id)
   843→   843→        async with self._lock:
   844→   844→            c = tenant.contradictions.get(contradiction_id)
   845→   845→            if not c:
   846→   846→                return False
   847→   847→            if c.status != ContradictionStatus.DETECTED:
   848→   848→                return False  # already resolved
   849→   849→            c.status = resolution
   850→   850→            c.resolved_at = now_utc()
   851→   851→            c.resolved_by = ctx.citizen_id
   852→   852→            c.resolution_notes = notes
   853→   853→
   854→   854→            # Apply resolution side-effects
   855→   855→            if resolution == ContradictionStatus.RESOLVED_KEEP_OLD:
   856→   856→                # Supersede the new memory (memory_b)
   857→   857→                entry_b = tenant.memories.get(c.memory_b_id)
   858→   858→                if entry_b:
   859→   859→                    mem_b, prov_b = entry_b
   860→   860→                    try:
   861→   861→                        validate_transition(mem_b.state, MemoryState.ARCHIVED)
   862→   862→                        mem_b.state = MemoryState.ARCHIVED
   863→   863→                        prov_b.append(
   864→   864→                            actor_id=ctx.citizen_id, action="archive",
   865→   865→                            process="contradiction_resolver",
   866→   866→                            notes=f"contradiction {contradiction_id} keep_old",
   867→   867→                            correlation_id=mem_b.metadata.correlation_id,
   868→   868→                        )
   869→   869→                        await self._persist(mem_b, prov_b)
   870→   870→                    except LifecycleError:
   871→   871→                        pass
   872→   872→            elif resolution == ContradictionStatus.RESOLVED_KEEP_NEW:
   873→   873→                entry_a = tenant.memories.get(c.memory_a_id)
   874→   874→                if entry_a:
   875→   875→                    mem_a, prov_a = entry_a
   876→   876→                    try:
   877→   877→                        validate_transition(mem_a.state, MemoryState.ARCHIVED)
   878→   878→                        mem_a.state = MemoryState.ARCHIVED
   879→   879→                        prov_a.append(
   880→   880→                            actor_id=ctx.citizen_id, action="archive",
   881→   881→                            process="contradiction_resolver",
   882→   882→                            notes=f"contradiction {contradiction_id} keep_new",
   883→   883→                            correlation_id=mem_a.metadata.correlation_id,
   884→   884→                        )
   885→   885→                        await self._persist(mem_a, prov_a)
   886→   886→                    except LifecycleError:
   887→   887→                        pass
   888→   888→
   889→   889→            self._observability.metrics.inc("memory.contradictions_resolved")
   890→   890→            await self._emit_event(
   891→   891→                "memory.contradiction_resolved", "",
   892→   892→                "", ctx.tenant_id,
   893→   893→                {
   894→   894→                    "contradiction_id": contradiction_id,
   895→   895→                    "resolution": resolution.value,
   896→   896→                    "actor_id": ctx.citizen_id,
   897→   897→                },
   898→   898→            )
   899→   899→            return True

   908→   908→        target_id: str,
   909→   909→        ctx: AuthorizationContext,

   914→   914→    ) -> AssociationMemory:
   915→   915→        """Create an association edge between two memories."""
   916→   916→        self._authz.authorize_capability(ctx, MemoryCapability.MEMORY_ASSOCIATE)
   917→   917→        tenant = self._get_or_create_tenant(ctx.tenant_id)
   918→   918→        async with self._lock:
   919→   919→            # Both memories must exist
   920→   920→            if source_id not in tenant.memories or target_id not in tenant.memories:
   921→   921→                raise ValidationError(
   922→   922→                    "Both source and target memories must exist for association"
   923→   923→                )
   924→   924→        edge = AssociationMemory.create(
   925→   925→            source_id=source_id, target_id=target_id,
   926→   926→            relationship=relationship, weight=weight,
   927→   927→            bidirectional=bidirectional,
   928→   928→            owner_id=ctx.citizen_id,
   929→   929→            tenant_id=ctx.tenant_id,
   930→   930→            correlation_id=new_correlation_id(),
   931→   931→        )
   932→   932→        # Build provenance
   933→   933→        prov = Provenance()
   934→   934→        prov.append(
   935→   935→            actor_id=ctx.citizen_id, action="associate",
   936→   936→            process="living_memory", notes=notes,
   937→   937→            correlation_id=edge.metadata.correlation_id,
   938→   938→        )
   939→   939→        # Run immune scan on the association memory
   940→   940→        report = self._immune.scan(
   941→   941→            memory=edge, provenance=prov,
   942→   942→            existing_ids=set(tenant.memories.keys()),
   943→   943→            expected_tenant_id=ctx.tenant_id,
   944→   944→        )
   945→   945→        if not report.accepted:
   946→   946→            raise ImmuneRejection(report.reason, report.to_dict())
   947→   947→        # Add to association graph
   948→   948→        graph = tenant.association_graph
   949→   949→        assoc_edge = AssociationEdge(
   950→   950→            source_id=source_id, target_id=target_id,
   951→   951→            relationship=relationship, weight=weight,
   952→   952→            bidirectional=bidirectional, created_by=ctx.citizen_id,
   953→   953→            correlation_id=edge.metadata.correlation_id,
   954→   954→        )
   955→   955→        await graph.add_edge(assoc_edge)
   956→   956→        # Persist association memory
   957→   957→        validate_transition(edge.state, MemoryState.ACTIVE)
   958→   958→        edge.state = MemoryState.ACTIVE
   959→   959→        async with self._lock:
   960→   960→            tenant.memories[edge.id.value] = (edge, prov)
   961→   961→        await self._persist(edge, prov)
   962→   962→        self._observability.metrics.inc("memory.associations")
   963→   963→        await self._emit_event(
   964→   964→            "memory.associated", edge.id.value,
   965→   965→            MemoryType.ASSOCIATIVE.value, ctx.tenant_id,
   966→   966→            {
   967→   967→                "actor_id": ctx.citizen_id,
   968→   968→                "source_id": source_id, "target_id": target_id,
   969→   969→                "relationship": relationship, "weight": weight,
   970→   970→            },
   971→   971→        )
   972→   972→        return edge
