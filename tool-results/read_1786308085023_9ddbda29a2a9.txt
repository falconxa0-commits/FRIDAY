     1→"""Action Ledger — approval gate with persistence and notifications.
     2→
     3→Key improvements over the original:
     4→- **asyncio.Event** replaces busy-polling in ``wait_for_approval()``.
     5→- Pending actions are persisted to a JSON file so they survive restarts.
     6→- A pluggable notification mechanism lets upstream code register
     7→  callbacks (e.g. push to WebSocket, send email) when actions are queued.
     8→- Audit log entries are **hash-chained** (tamper-evident). Any
     9→  modification to a past entry is detectable via ``verify_chain()``.
    10→"""
    11→
    12→import asyncio
    13→import hashlib
    14→import hmac
    15→import json
    16→import os
    17→import uuid
    18→import datetime
    19→import logging
    20→from typing import Callable, Dict, List, Optional
    21→
    22→from config.settings import AUTONOMY_PROFILE
    23→
    24→logger = logging.getLogger(__name__)
    25→
    26→
    27→# Components that must NEVER be auto-approved regardless of autonomy profile.
    28→# These involve physical-world effects, financial cost, or irreversible actions.
    29→NEVER_AUTO_APPROVE_COMPONENTS = frozenset({
    30→    "ImageGen",        # Costs money per generation
    31→    "VideoGen",        # Costs money per generation
    32→    "CodeExecution",   # Arbitrary code execution risk
    33→    "Printer",         # Physical world effect
    34→    "Printer3D",       # Physical world effect + material cost
    35→    "Finance",         # Financial transactions — money movement
    36→    "Commerce",        # Purchase / checkout actions

    40→class ActionLedger:
    41→    PERSIST_PATH = "action_ledger_pending.json"
    42→
    43→    def __init__(self):
    44→        self.pending_actions: Dict[str, dict] = {}
    45→        self.audit_log = "action_ledger_audit.log"
    46→        self.profile = AUTONOMY_PROFILE
    47→
    48→        # In-memory hash-chained audit log (each entry includes a hash
    49→        # that depends on the previous entry's hash — any tampering
    50→        # breaks the chain and is detectable via verify_chain()).
    51→        # The chain is persisted to CHAIN_PERSIST_PATH so tamper-evidence
    52→        # survives process restarts.
    53→        self._audit_chain: List[dict] = []
    54→
    55→        # Per-action events for efficient await (replaces busy-polling)
    56→        self._events: Dict[str, asyncio.Event] = {}
    57→
    58→        # Notification callbacks — called when a new action is queued
    59→        self._notification_callbacks: list[Callable[[dict], None]] = []
    60→
    61→        # Restore previously persisted pending actions
    62→        self._load_persisted()
    63→        # Restore persisted hash-chained audit log
    64→        self._load_chain()

    70→    def _persist(self):
    71→        """Write current pending actions to disk (fire-and-forget best-effort)."""
    72→        try:
    73→            with open(self.PERSIST_PATH, "w") as f:
    74→                json.dump(self.pending_actions, f, indent=2, default=str)
    75→        except Exception:
    76→            logger.exception("Failed to persist action ledger")
    77→
    78→    def _load_persisted(self):
    79→        """Load previously persisted pending actions from disk."""
    80→        if not os.path.exists(self.PERSIST_PATH):
    81→            return
    82→        try:
    83→            with open(self.PERSIST_PATH, "r") as f:
    84→                data = json.load(f)
    85→            if isinstance(data, dict):
    86→                for action_id, action_data in data.items():
    87→                    if action_data.get("status") == "pending":
    88→                        self.pending_actions[action_id] = action_data
    89→                        self._events[action_id] = asyncio.Event()
    90→                logger.info(
    91→                    "Restored %d pending actions from persistence", len(self.pending_actions)
    92→                )
    93→        except Exception:
    94→            logger.exception("Failed to load persisted action ledger")
    95→

   100→    def register_notification_callback(self, callback: Callable[[dict], None]):
   101→        """Register a callback invoked when a new action is queued.
   102→
   103→        The callback receives the action dict and can, for example,
   104→        push it over a WebSocket or send an email.
   105→        """
   106→        self._notification_callbacks.append(callback)

   108→    def _notify(self, action_data: dict):
   109→        for cb in self._notification_callbacks:
   110→            try:
   111→                cb(action_data)
   112→            except Exception:
   113→                logger.exception("Notification callback raised an error")

   116→    # Audit logging (hash-chained / tamper-evident)

   119→    GENESIS_HASH = "genesis"
   120→    CHAIN_PERSIST_PATH = "action_ledger_chain.json"
   121→
   122→    # Keys in action params that are redacted in the file-based audit log
   123→    # (the hash chain still includes the full params — only the human-readable
   124→    # text log is redacted, to prevent secret leakage into log aggregators).
   125→    SENSITIVE_PARAM_KEYS = frozenset({
   126→        "password", "passwd", "pwd",
   127→        "api_key", "apikey", "token", "secret",
   128→        "payment_method", "card_number", "cvv", "expiry",
   129→        "client_secret", "access_token", "refresh_token",
   130→        "stripe_token", "payment_intent_id",
   131→    })
   132→
   133→    @staticmethod
   134→    def _redact_params(params: dict) -> dict:
   135→        """Return a copy of params with sensitive values replaced by '***'.
   136→
   137→        Used only for the human-readable file log. The hash chain
   138→        includes the full params so tamper detection still works.
   139→        """
   140→        if not isinstance(params, dict):
   141→            return params
   142→        redacted = {}
   143→        for k, v in params.items():
   144→            if k.lower() in ActionLedger.SENSITIVE_PARAM_KEYS:
   145→                redacted[k] = "***REDACTED***"
   146→            elif isinstance(v, str) and len(v) > 100:
   147→                redacted[k] = v[:50] + "...(truncated)"
   148→            else:
   149→                redacted[k] = v
   150→        return redacted

   152→    # Server-side secret for HMAC-SHA256. Loaded once at class init.
   153→    # Prevents offline hash forgery: an attacker who modifies the JSON
   154→    # file cannot recompute valid hashes without this secret.
   155→    # Sourced from FRIDAY_LEDGER_HMAC_SECRET env var, or derived from
   156→    # FRIDAY_API_TOKEN, or generated once and persisted to ~/.friday/ledger_secret.
   157→    _HMAC_SECRET: Optional[str] = None
   158→
   159→    @classmethod
   160→    def _get_hmac_secret(cls) -> bytes:
   161→        """Return the HMAC secret, loading it lazily on first use.
   162→
   163→        Priority:
   164→            1. FRIDAY_LEDGER_HMAC_SECRET env var (operator-set)
   165→            2. FRIDAY_API_TOKEN env var (already a secure random)
   166→            3. Persisted random secret at ~/.friday/ledger_secret
   167→               (auto-generated on first run, chmod 600)
   168→
   169→        The secret is bytes-encoded UTF-8. Returns a non-empty bytestring.
   170→        """
   171→        if cls._HMAC_SECRET is not None:
   172→            return cls._HMAC_SECRET.encode("utf-8")
   173→
   174→        env_secret = os.environ.get("FRIDAY_LEDGER_HMAC_SECRET")
   175→        if env_secret and env_secret.strip():
   176→            cls._HMAC_SECRET = env_secret.strip()
   177→            return cls._HMAC_SECRET.encode("utf-8")
   178→
   179→        api_token = os.environ.get("FRIDAY_API_TOKEN", "")
   180→        if api_token and api_token.strip() and not api_token.startswith("your_"):
   181→            cls._HMAC_SECRET = api_token.strip()
   182→            return cls._HMAC_SECRET.encode("utf-8")
   183→
   184→        # Persisted secret path — generated once, reused across restarts
   185→        secret_path = os.path.expanduser("~/.friday/ledger_secret")
   186→        try:
   187→            if os.path.exists(secret_path):
   188→                with open(secret_path, "r") as f:
   189→                    cls._HMAC_SECRET = f.read().strip()
   190→                    if cls._HMAC_SECRET:
   191→                        return cls._HMAC_SECRET.encode("utf-8")
   192→            # Generate a new 32-byte URL-safe secret
   193→            import secrets as _secrets
   194→            os.makedirs(os.path.dirname(secret_path), exist_ok=True)
   195→            cls._HMAC_SECRET = _secrets.token_urlsafe(32)
   196→            with open(secret_path, "w") as f:
   197→                f.write(cls._HMAC_SECRET)
   198→            try:
   199→                os.chmod(secret_path, 0o600)
   200→            except OSError:
   201→                pass  # best-effort on non-POSIX systems
   202→            return cls._HMAC_SECRET.encode("utf-8")
   203→        except Exception:
   204→            # Last-resort fallback: derive from hostname + username.
   205→            # NOT cryptographically strong, but better than no HMAC.
   206→            import getpass
   207→            import socket
   208→            fallback = f"friday-fallback-{socket.gethostname()}-{getpass.getuser()}"
   209→            cls._HMAC_SECRET = fallback
   210→            return cls._HMAC_SECRET.encode("utf-8")
   211→
   212→    @staticmethod
   213→    def _compute_entry_hash(entry: dict, prev_hash: str) -> str:
   214→        """HMAC-SHA256 of (prev_hash || action_id || component || action ||
   215→        params || timestamp || status || approved_by).
   216→
   217→        All seven fields are included in the hash content so that any
   218→        modification — including rewriting WHO approved an action —
   219→        produces a different hash and breaks the chain.
   220→
   221→        Uses HMAC-SHA256 with a server-side secret rather than bare
   222→        SHA-256, so an attacker with read access to the JSON file
   223→        cannot recompute valid hashes offline.
   224→
   225→        Backward-compat: if the HMAC secret is unavailable (e.g., very
   226→        old install pre-upgrade), falls back to bare SHA-256 of the
   227→        same seven fields — still catches tampering of any field
   228→        except via offline recomputation.
   229→        """
   230→        content = json.dumps({
   231→            "prev_hash": prev_hash,
   232→            "action_id": entry.get("action_id") or entry.get("id", ""),
   233→            "component": entry.get("component", ""),
   234→            "action": entry.get("action", ""),
   235→            "params": entry.get("params", {}),
   236→            "timestamp": entry.get("timestamp", ""),
   237→            "status": entry.get("status", ""),
   238→            "approved_by": entry.get("approved_by", ""),
   239→        }, sort_keys=True, default=str)
   240→
   241→        try:
   242→            secret = ActionLedger._get_hmac_secret()
   243→            return hmac.new(secret, content.encode("utf-8"), hashlib.sha256).hexdigest()
   244→        except Exception:
   245→            # Fallback: bare SHA-256 (still includes approved_by, so
   246→            # the forgery is caught; only offline recomputation is possible)
   247→            return hashlib.sha256(content.encode("utf-8")).hexdigest()
   248→
   249→    def _log_audit(self, action_data, approved_by="human"):
   250→        """Append an entry to BOTH the file-based audit log and the
   251→        persisted hash-chained audit log.
   252→
   253→        - File log: human-readable, append-only, with SENSITIVE params redacted.
   254→        - Hash chain: persisted to ``CHAIN_PERSIST_PATH`` (JSON file),
   255→          tamper-evident across process restarts.
   256→        """
   257→        # 1. File-based log (human-readable, append-only, REDACTED)
   258→        try:
   259→            redacted_params = self._redact_params(action_data.get("params", {}))
   260→            with open(self.audit_log, "a") as f:
   261→                f.write(
   262→                    f"{datetime.datetime.now().isoformat()} | "
   263→                    f"{approved_by.upper()} | "
   264→                    f"{action_data['component']}.{action_data['action']} | "
   265→                    f"{redacted_params}\n"
   266→                )
   267→        except Exception:
   268→            logger.exception("Failed to write audit log file")
   269→
   270→        # 2. Hash-chained log (tamper-evident, PERSISTED to disk)
   271→        prev_hash = self._audit_chain[-1]["hash"] if self._audit_chain else self.GENESIS_HASH
   272→        entry = {
   273→            "action_id": action_data.get("id", ""),
   274→            "component": action_data.get("component", ""),
   275→            "action": action_data.get("action", ""),
   276→            "params": action_data.get("params", {}),  # full params for hash integrity
   277→            "timestamp": action_data.get("timestamp", datetime.datetime.now().isoformat()),
   278→            "status": action_data.get("status", ""),
   279→            "approved_by": approved_by,
   280→            "prev_hash": prev_hash,
   281→        }
   282→        entry["hash"] = self._compute_entry_hash(entry, prev_hash)
   283→        self._audit_chain.append(entry)
   284→        # Persist the full chain to disk so tamper-evidence survives restarts
   285→        self._persist_chain()
   286→
   287→    def _persist_chain(self) -> None:
   288→        """Persist the hash-chained audit log to disk as JSON.
   289→
   290→        Each entry's hash is recomputed from its contents, so modifying
   291→        the JSON file and restarting will be detected by verify_chain().
   292→        """
   293→        try:
   294→            with open(self.CHAIN_PERSIST_PATH, "w") as f:
   295→                json.dump(self._audit_chain, f, indent=2, default=str)
   296→        except Exception:
   297→            logger.exception("Failed to persist audit chain")
   298→
   299→    def _load_chain(self) -> None:
   300→        """Load the persisted hash-chained audit log from disk on startup.
   301→
   302→        If the chain is broken (tamper detected), the on-disk file is
   303→        archived to ``<path>.tampered.<timestamp>.json`` for forensic
   304→        analysis rather than silently discarded. The in-memory chain
   305→        starts fresh from genesis.
   306→        """
   307→        import os as _os
   308→        if not _os.path.exists(self.CHAIN_PERSIST_PATH):
   309→            return
   310→        try:
   311→            with open(self.CHAIN_PERSIST_PATH, "r") as f:
   312→                data = json.load(f)
   313→            if isinstance(data, list):
   314→                # Validate the loaded chain before accepting it
   315→                prev_hash = self.GENESIS_HASH
   316→                valid = True
   317→                for entry in data:
   318→                    expected = self._compute_entry_hash(entry, prev_hash)
   319→                    if entry.get("hash") != expected:
   320→                        logger.warning(
   321→                            "Loaded audit chain has broken hash at entry — "
   322→                            "archiving chain for forensics (possible tampering)."
   323→                        )
   324→                        valid = False
   325→                        # Archive the tampered file instead of silently discarding
   326→                        try:
   327→                            archive_path = (
   328→                                f"{self.CHAIN_PERSIST_PATH}.tampered."
   329→                                f"{datetime.datetime.now().strftime('%Y%m%d_%H%M%S')}.json"
   330→                            )
   331→                            with open(archive_path, "w") as af:
   332→                                json.dump(data, af, indent=2, default=str)
   333→                            logger.warning(
   334→                                "Archived tampered audit chain to %s", archive_path
   335→                            )
   336→                        except Exception as archive_exc:
   337→                            logger.error(
   338→                                "Failed to archive tampered chain: %s", archive_exc
   339→                            )
   340→                        break
   341→                    prev_hash = entry.get("hash", "")
   342→                if valid:
   343→                    self._audit_chain = data
   344→                    logger.info(
   345→                        "Loaded %d entries from persisted audit chain",
   346→                        len(self._audit_chain),
   347→                    )
   348→                else:
   349→                    # Start fresh but preserve the archived copy
   350→                    self._audit_chain = []
   351→                    logger.warning(
   352→                        "Starting with a fresh audit chain. The tampered "
   353→                        "chain was preserved on disk for forensic analysis."
   354→                    )
   355→        except Exception:
   356→            logger.exception("Failed to load persisted audit chain")
   357→
   358→    def get_audit_log(self) -> List[dict]:
   359→        """Return the hash-chained audit log entries (newest last)."""
   360→        return list(self._audit_chain)
   361→
   362→    def verify_chain(self) -> bool:
   363→        """Verify the integrity of the hash-chained audit log.
   364→
   365→        Returns True if every entry's hash matches the recomputed hash
   366→        and every prev_hash link is intact. Returns False if any entry
   367→        was tampered with.
   368→        """
   369→        prev_hash = self.GENESIS_HASH
   370→        for i, entry in enumerate(self._audit_chain):
   371→            expected = self._compute_entry_hash(entry, prev_hash)
   372→            if entry.get("hash") != expected:
   373→                logger.warning(
   374→                    "Ledger chain BROKEN at index %d: hash mismatch "
   375→                    "(expected %s, got %s)", i, expected, entry.get("hash")
   376→                )
   377→                return False
   378→            if entry.get("prev_hash") != prev_hash:
   379→                logger.warning(
   380→                    "Ledger chain BROKEN at index %d: prev_hash link mismatch", i
   381→                )
   382→                return False
   383→            prev_hash = entry["hash"]
   384→        return True

   386→    def _tamper_for_test(self, index: int, new_params: dict) -> None:
   387→        """Intentionally modify an audit entry's params (TEST ONLY).
   388→
   389→        This breaks the hash chain. ``verify_chain()`` will return False
   390→        afterwards. Used only by the verification script to demonstrate
   391→        that tampering is detectable.
   392→        """
   393→        if 0 <= index < len(self._audit_chain):
   394→            self._audit_chain[index]["params"] = new_params
   395→            # NOTE: We do NOT recompute the hash — that's the point.

   401→    def _should_auto_approve(self, component, risk_level):
   402→        # Never auto-approve dangerous components regardless of profile
   403→        if component in NEVER_AUTO_APPROVE_COMPONENTS:
   404→            return False
   405→        if self.profile == "GUEST":
   406→            return False
   407→        if self.profile == "STANDARD":
   408→            if risk_level == "low":
   409→                return True
   410→        if self.profile == "POWER":
   411→            if component in ("PCControl", "BrowserControl") and risk_level != "critical":
   412→                return True
   413→            if risk_level == "low":
   414→                return True
   415→        return False

   417→    def queue_action(self, component, action, params, risk_level="high"):
   418→        action_id = str(uuid.uuid4())
   419→        action_data = {
   420→            "id": action_id,
   421→            "component": component,
   422→            "action": action,
   423→            "params": params,
   424→            "risk_level": risk_level,
   425→            "status": "pending",
   426→            "timestamp": datetime.datetime.now().isoformat(),
   427→        }
   428→
   429→        # Auto-approve based on profile
   430→        if self._should_auto_approve(component, risk_level):
   431→            action_data["status"] = "approved"
   432→            self._log_audit(action_data, approved_by="auto")
   433→            self.pending_actions[action_id] = action_data
   434→            self._events[action_id] = asyncio.Event()
   435→            self._events[action_id].set()  # already approved
   436→            self._persist()
   437→            return action_id
   438→
   439→        self.pending_actions[action_id] = action_data
   440→        self._events[action_id] = asyncio.Event()
   441→        self._persist()
   442→        self._notify(action_data)
   443→        return action_id

   445→    async def wait_for_approval(self, action_id, timeout=300):
   446→        """Wait for approval using ``asyncio.Event`` (no busy-polling).
   447→
   448→        Returns ``True`` if approved, ``False`` if rejected or timed out.
   449→        """
   450→        event = self._events.get(action_id)
   451→        if event is None:
   452→            return False
   453→
   454→        # Already approved?
   455→        if self.pending_actions.get(action_id, {}).get("status") == "approved":
   456→            return True
   457→
   458→        logger.info(
   459→            "Action %s awaiting manual approval (Profile: %s)",
   460→            action_id,
   461→            self.profile,
   462→        )
   463→
   464→        try:
   465→            await asyncio.wait_for(event.wait(), timeout=timeout)
   466→        except asyncio.TimeoutError:
   467→            return False
   468→
   469→        status = self.pending_actions.get(action_id, {}).get("status")
   470→        if status == "approved":
   471→            self._log_audit(self.pending_actions[action_id], approved_by="human")
   472→            return True
   473→        return False
   474→
   475→    def approve_action(self, action_id):
   476→        if action_id in self.pending_actions:
   477→            self.pending_actions[action_id]["status"] = "approved"
   478→            event = self._events.get(action_id)
   479→            if event:
   480→                event.set()
   481→            self._persist()
   482→            # Add to the hash-chained audit log
   483→            self._log_audit(self.pending_actions[action_id], approved_by="human")
   484→            return True
   485→        return False
   486→
   487→    def reject_action(self, action_id):
   488→        if action_id in self.pending_actions:
   489→            self.pending_actions[action_id]["status"] = "rejected"
   490→            event = self._events.get(action_id)
   491→            if event:
   492→                event.set()  # unblock waiters (they'll check status)
   493→            self._persist()
   494→            # Add to the hash-chained audit log
   495→            self._log_audit(self.pending_actions[action_id], approved_by="human_rejected")
   496→            return True
   497→        return False

   501→    # ----------------------------------------------------------------
   502→
   503→    async def wait_for_voice_approval(
   504→        self,
   505→        action_id: str,
   506→        speaker=None,
   507→        listener=None,
   508→        timeout: int = 60,
   509→        max_retries: int = 1,
   510→    ) -> bool:
   511→        """Speak the pending action aloud and wait for a spoken yes/no.
   512→
   513→        Workflow:
   514→            1. Speak the action description via the speaker module.
   515→            2. Use the listener to capture a spoken response (bounded by
   516→               ``timeout`` seconds; if no audio is captured in that window
   517→               we treat it as "no response").
   518→            3. Transcribe via ``voice.transcriber`` and parse intent:
   519→                 yes -> approve_action() and confirm aloud
   520→                 no  -> reject_action() and confirm aloud
   521→                 unclear -> ask once more, then default to reject
   522→            4. On timeout (no audio captured) -> reject and say so aloud.
   523→
   524→        Args:
   525→            action_id: The queued action to approve or reject.
   526→            speaker: FridaySpeaker instance (or any object with ``speak``).
   527→            listener: FridayListener instance (or any object with ``record_audio``).
   528→            timeout: Seconds to wait for a response (default 60).
   529→            max_retries: How many times to re-ask on ambiguous responses.
   530→
   531→        Returns:
   532→            True if approved, False if rejected or timed out.
   533→        """
   534→        action_data = self.pending_actions.get(action_id)
   535→        if not action_data:
   536→            logger.warning("wait_for_voice_approval: unknown action %s", action_id)
   537→            return False
   538→
   539→        # Already approved?
   540→        if action_data.get("status") == "approved":
   541→            return True
   542→
   543→        # Build a spoken description
   544→        component = action_data.get("component", "unknown")
   545→        act = action_data.get("action", "unknown")
   546→        params = action_data.get("params", {})
   547→        description = (
   548→            f"Action pending: {component} wants to {act}. "
   549→            f"Parameters: {params}. "
   550→            f"Say yes to approve, or no to reject."
   551→        )
   552→
   553→        async def _speak(text: str) -> None:
   554→            if speaker and hasattr(speaker, "speak"):
   555→                try:
   556→                    await speaker.speak_async(text)
   557→                    return
   558→                except Exception as exc:
   559→                    logger.warning("Voice approval: speak_async failed: %s", exc)
   560→                    try:
   561→                        speaker.speak(text)
   562→                        return
   563→                    except Exception as e:
   564→                        logger.debug(f"Non-critical error: {e}")
   565→            print(f"[VOICE APPROVAL] {text}")
   566→
   567→        async def _listen_once() -> str:
   568→            """Capture one audio chunk and return the transcribed text.
   569→
   570→            Returns "" if no audio was captured or transcription failed.
   571→            Bounds the listen by ``timeout`` seconds.
   572→            """
   573→            if not (listener and hasattr(listener, "record_audio")):
   574→                return ""
   575→
   576→            import tempfile
   577→            import os as _os
   578→
   579→            tmp_path = None
   580→            try:
   581→                with tempfile.NamedTemporaryFile(suffix=".wav", delete=False) as tmp:
   582→                    tmp_path = tmp.name
   583→
   584→                recorded = await asyncio.wait_for(
   585→                    asyncio.to_thread(listener.record_audio, tmp_path, 8),
   586→                    timeout=timeout,
   587→                )
   588→                if not recorded:
   589→                    return ""
   590→
   591→                try:
   592→                    from voice.transcriber import FridayTranscriber
   593→                    transcriber = FridayTranscriber()
   594→                    text = await asyncio.to_thread(transcriber.transcribe, tmp_path)
   595→                    return text or ""
   596→                except ImportError:
   597→                    logger.warning(
   598→                        "Voice approval: whisper not installed - cannot transcribe"
   599→                    )
   600→                    return ""
   601→                except Exception as exc:
   602→                    logger.warning("Voice approval: transcription failed: %s", exc)
   603→                    return ""
   604→            except asyncio.TimeoutError:
   605→                logger.info(
   606→                    "Voice approval: no response within %ss - treating as no input",
   607→                    timeout,
   608→                )
   609→                return ""
   610→            finally:
   611→                if tmp_path:
   612→                    try:
   613→                        _os.unlink(tmp_path)
   614→                    except OSError as e:
   615→                        logger.debug(f"Non-critical error: {e}")
   616→
   617→        # Step 1: Speak the action
   618→        await _speak(description)
   619→
   620→        # Step 2 + 3: Listen and parse, with retry for ambiguous responses
   621→        attempts = 0
   622→        approved = None
   623→        while attempts <= max_retries:
   624→            attempts += 1
   625→            response_text = await _listen_once()
   626→
   627→            if not response_text:
   628→                if attempts == 1:
   629→                    await _speak(
   630→                        "No response detected within the timeout window. "
   631→                        "Rejecting the action for safety."
   632→                    )
   633→                approved = False
   634→                break
   635→
   636→            approved = self._parse_voice_intent(response_text)
   637→            logger.info(
   638→                "Voice approval attempt %d: heard=%r parsed=%r",
   639→                attempts, response_text, approved,
   640→            )
   641→
   642→            if approved is True:
   643→                await _speak("Approved. Proceeding.")
   644→                break
   645→            if approved is False:
   646→                await _speak("Rejected. The action will not run.")
   647→                break
   648→
   649→            if attempts <= max_retries:
   650→                await _speak("I didn\'t catch that. Please say yes or no.")
   651→                continue
   652→            await _speak("Still unclear. Rejecting the action for safety.")
   653→            approved = False
   654→            break
   655→
   656→        # Step 4: Approve or reject
   657→        if approved:
   658→            self.approve_action(action_id)
   659→            self._log_audit(action_data, approved_by="voice")
   660→        else:
   661→            self.reject_action(action_id)
   662→            self._log_audit(action_data, approved_by="voice_rejected")
   663→
   664→        return bool(approved)
   665→
   666→    @staticmethod
   667→    def _parse_voice_intent(text: str):
   668→        """Parse a spoken response into yes / no / None (ambiguous).
   669→
   670→        Returns:
   671→            True  — positive intent detected
   672→            False — negative intent detected
   673→            None  — ambiguous / unclear
   674→        """
   675→        if not text:
   676→            return None
   677→
   678→        text_lower = text.lower().strip()
   679→
   680→        yes_words = {"yes", "yeah", "yep", "sure", "okay", "ok", "approve", "go ahead", "do it", "confirm", "affirmative"}
   681→        no_words = {"no", "nope", "nah", "reject", "deny", "cancel", "stop", "don't", "negative", "refuse"}
   682→
   683→        for word in yes_words:
   684→            if word in text_lower:
   685→                return True
   686→
   687→        for word in no_words:
   688→            if word in text_lower:
   689→                return False
   690→
   691→        return None
