   261→
   262→class FridayMCPServer:
   263→    """FRIDAY MCP Server — wraps the action layer with ledger/auth/receipt."""
   264→
   265→    def __init__(self):
   266→        self._glm_brain = None
   267→        self._ledger = None
   268→
   269→    def _get_glm_brain(self):
   270→        """Lazy-initialize GLMBrain."""
   271→        if self._glm_brain is None:
   272→            try:
   273→                from core.glm_brain import GLMBrain
   274→                self._glm_brain = GLMBrain()
   275→            except Exception as exc:
   276→                logger.error(f"Failed to initialize GLMBrain: {exc}")
   277→        return self._glm_brain
   278→
   279→    def _get_ledger(self):
   280→        """Lazy-initialize the action ledger."""
   281→        if self._ledger is None:
   282→            try:
   283→                from core.ledger import get_ledger
   284→                self._ledger = get_ledger()
   285→            except Exception as exc:
   286→                logger.error(f"Failed to initialize ledger: {exc}")
   287→        return self._ledger
   288→
   289→    def _make_receipt(self, action: str, result: dict) -> dict:
   290→        """Create a receipt for an MCP tool call."""
   291→        return {
   292→            "action": action,
   293→            "timestamp": datetime.datetime.now().isoformat(),
   294→            "status": result.get("status", "unknown"),
   295→            "data": result.get("receipt", result.get("message", "")),
   296→        }
   297→
   298→    def _compute_risk_level(
   299→        self, component: str, action: str, params: dict
   300→    ) -> str:
   301→        """Compute the risk level for an action using EthicalSentinel.
   302→
   303→        SECURITY: This is the single source of truth for risk
   304→        classification on the MCP layer. Caller-supplied risk_level
   305→        is always ignored. Falls back to UniversalConnector._classify_risk
   306→        if the Sentinel is unavailable.
   307→
   308→        Returns one of: "low", "medium", "high", "critical".
   309→        """
   310→        # Try EthicalSentinel first (deeper analysis)
   311→        try:
   312→            from core.sentinel import EthicalSentinel
   313→            sentinel = EthicalSentinel()
   314→            context = {
   315→                "component": component,
   316→                "params": params or {},
   317→            }
   318→            result = sentinel.evaluate_action(action, context=context)
   319→            classification = result.get("classification", "")
   320→            if isinstance(classification, str):
   321→                classification = classification.upper()
   322→            elif hasattr(classification, "value"):
   323→                classification = classification.value.upper()
   324→            else:
   325→                classification = str(classification).upper()
   326→
   327→            mapping = {
   328→                "SAFE": "low",
   329→                "CAUTIOUS": "medium",
   330→                "DANGEROUS": "high",
   331→                "CRITICAL": "critical",
   332→            }
   333→            mapped = mapping.get(classification)
   334→            if mapped:
   335→                return mapped
   336→        except Exception as exc:
   337→            logger.debug(f"Sentinel unavailable, falling back to keyword classifier: {exc}")
   338→
   339→        # Fallback: keyword-based classifier from UniversalConnector
   340→        try:
   341→            from core.universal_connector import UniversalConnector
   342→            return UniversalConnector._classify_risk(action)
   343→        except Exception:
   344→            # Last resort: assume high risk
   345→            return "high"

   349→    async def handle_chat(self, params: dict) -> dict:
   350→        """Handle the 'chat' tool — logged in audit trail (read-only, no ledger gate needed)."""
   351→        message = params.get("message", "")
   352→        if not message:
   353→            return {"status": "error", "message": "No message provided."}
   354→
   355→        # Log in audit trail (read-only calls don't need approval gate,
   356→        # but must be logged for transparency)
   357→        ledger = self._get_ledger()
   358→        if ledger:
   359→            try:
   360→                action_id = ledger.queue_action(
   361→                    "MCPChat", "chat", {"message_length": len(message)},
   362→                    risk_level="low",
   363→                )
   364→                # Read-only calls auto-approve at STANDARD+ profile
   365→                ledger.approve_action(action_id)
   366→            except Exception as e:
   367→                logger.debug(f"Non-critical error: {e}")
   368→
   369→        glm = self._get_glm_brain()
   370→        if glm and glm.available():
   371→            full_response = ""
   372→            async for chunk in glm.chat_stream(message):
   373→                full_response += chunk
   374→            return {
   375→                "status": "success",
   376→                "message": full_response,
   377→                "receipt": self._make_receipt("chat", {"status": "success", "receipt": {
   378→                    "provider": "glm",
   379→                    "response_length": len(full_response),
   380→                }}),
   381→            }
   382→
   383→        # Fallback to FridayBrain
   384→        try:
   385→            from core.brain import FridayBrain
   386→            brain = FridayBrain()
   387→            full_response = ""
   388→            async for chunk in brain.chat_stream(message):
   389→                full_response += chunk
   390→            return {
   391→                "status": "success",
   392→                "message": full_response,
   393→                "receipt": self._make_receipt("chat", {"status": "success", "receipt": {
   394→                    "provider": brain.get_stats().get("provider", "unknown"),
   395→                    "response_length": len(full_response),
   396→                }}),
   397→            }
   398→        except Exception as exc:
   399→            return {"status": "error", "message": str(exc)}

   407→            return {"status": "error", "message": "No image provided."}

   409→        # Log in audit trail
   410→        ledger = self._get_ledger()
   411→        if ledger:
   412→            try:
   413→                action_id = ledger.queue_action(
   414→                    "MCPVision", "vision_analyze", {"prompt": prompt[:100]},

   470→    async def handle_image_generation(self, params: dict) -> dict:
   471→        """Handle the 'image_generation' tool — gated through ledger."""
   472→        prompt = params.get("prompt", "")
   473→        size = params.get("size", "1024x1024")
   474→
   475→        if not prompt:

   478→        # Gate through ledger — FAIL CLOSED if ledger unavailable
   479→        ledger = self._get_ledger()
   480→        if ledger is None:
   481→            return {
   482→                "status": "error",
   483→                "message": "Ledger unavailable — refusing to execute without approval gate.",
   484→                "receipt": self._make_receipt("image_generation", {"status": "error"}),

   487→            "ImageGen", "generate_image",

   489→            risk_level="high",
   490→        )

   494→        if not approved:
   495→            return {
   496→                "status": "error",
   497→                "message": "Image generation requires explicit approval. Action queued but not approved within timeout.",
   498→                "receipt": self._make_receipt("image_generation", {"status": "rejected"}),
   499→            }
   500→
   501→        try:
   502→            from integrations.image_gen import ImageGen
   503→            ig = ImageGen()
   504→            result = await ig.execute("generate_image", {"prompt": prompt, "size": size})
   505→            return {
   506→                "status": result.get("status", "unknown"),
   507→                "message": result.get("message", ""),
   508→                "receipt": self._make_receipt("image_generation", result),
   509→            }
   510→        except Exception as exc:
   511→            return {"status": "error", "message": str(exc)}

   594→    async def handle_execute_action(self, params: dict) -> dict:
   595→        """Handle 'friday.execute_action' — routes to UniversalConnector.
   596→
   597→        All actions go through the ledger's NEVER_AUTO_APPROVE_COMPONENTS
   598→        gate. Physical/financial/code-execution actions require explicit
   599→        user approval regardless of autonomy profile.
   600→        """
   601→        service = params.get("service") or params.get("component")
   602→        action = params.get("action")
   603→        action_params = params.get("params") or params.get("args") or {}
   604→
   605→        if not service or not action:
   606→            return {
   607→                "status": "error",
   608→                "message": "Missing required parameters: service and action",
   609→            }
   610→
   611→        try:
   612→            from core.universal_connector import UniversalConnector
   613→            connector = UniversalConnector()
   614→            # Use a short approval timeout so MCP clients get a clear
   615→            # "pending approval" response quickly rather than hanging.
   616→            result = await connector.execute_action(
   617→                service, action, action_params, approval_timeout=5,
   618→            )
   619→            return {
   620→                "status": result.get("status", "unknown"),
   621→                "message": result.get("message", ""),
   622→                "receipt": self._make_receipt(
   623→                    f"execute_action:{service}.{action}", result,
   624→                ),
   625→            }
   626→        except Exception as exc:
   627→            return {
   628→                "status": "error",
   629→                "message": f"execute_action failed: {exc}",
   630→                "receipt": self._make_receipt("execute_action", {"status": "error"}),
   631→            }
   632→
   633→    async def handle_request_approval(self, params: dict) -> dict:
   634→        """Handle 'friday.request_approval' — external agent submits an action
   635→        for human approval via Friday's ledger.
   636→
   637→        This is the killer feature for trusted execution: another AI agent
   638→        (Claude Code, Cursor, etc.) can say "I want to delete this file"
   639→        and Friday's human-in-the-loop gate stands between the suggestion
   640→        and the action.
   641→
   642→        Friday speaks the request out loud (if voice is configured), waits
   643→        for the human's real voice/keyboard response, and returns
   644→        approved/rejected to the calling agent.
   645→
   646→        Params:
   647→            component: The integration component (e.g. "Filesystem")
   648→            action: The action name (e.g. "delete_file")
   649→            params: Action parameters
   650→            risk_level: "low" | "medium" | "high" | "critical"
   651→            description: Human-readable description of what the agent wants to do
   652→            timeout: Seconds to wait for approval (default 60)
   653→            use_voice: If True, use voice-native approval (default False)
   654→
   655→        Returns:
   656→            {status: "approved" | "rejected" | "timeout",
   657→             action_id: str,
   658→             receipt: dict}
   659→        """
   660→        component = params.get("component")
   661→        action = params.get("action")
   662→        action_params = params.get("params") or {}
   663→        # SECURITY: Ignore caller-supplied risk_level. An external agent
   664→        # (Claude Code, Cursor) should not be able to self-classify a
   665→        # destructive action as "low" to bypass the approval gate.
   666→        # We compute risk_level ourselves via EthicalSentinel (or the
   667→        # connector's _classify_risk fallback) using the component +
   668→        # action + params.
   669→        caller_risk_level = params.get("risk_level")
   670→        if caller_risk_level is not None:
   671→            logger.warning(
   672→                "MCP request_approval: caller supplied risk_level=%r — "
   673→                "IGNORING (security policy). Risk will be computed by "
   674→                "Friday's EthicalSentinel.",
   675→                caller_risk_level,
   676→            )
   677→        risk_level = self._compute_risk_level(component, action, action_params)
   678→        description = params.get("description", f"{component}.{action}")
   679→        timeout = int(params.get("timeout", 60))
   680→        use_voice = bool(params.get("use_voice", False))
   681→
   682→        if not component or not action:
   683→            return {
   684→                "status": "error",
   685→                "message": "Missing required parameters: component and action",
   686→                "receipt": self._make_receipt("request_approval", {"status": "error"}),
   687→            }
   688→
   689→        ledger = self._get_ledger()
   690→        if not ledger:
   691→            return {
   692→                "status": "error",
   693→                "message": "Ledger not available",
   694→                "receipt": self._make_receipt("request_approval", {"status": "error"}),
   695→            }
   696→
   697→        # Queue the action in the ledger (this is the gate)
   698→        action_id = ledger.queue_action(
   699→            component, action, action_params, risk_level=risk_level,
   700→        )
   701→        logger.info(
   702→            "MCP request_approval: queued %s.%s (id=%s, risk=%s, timeout=%ss)",
   703→            component, action, action_id, risk_level, timeout,
   704→        )
   705→
   706→        # Wait for approval
   707→        try:
   708→            if use_voice:
   709→                approved = await ledger.wait_for_voice_approval(
   710→                    action_id, timeout=timeout,
   711→                )
   712→            else:
   713→                approved = await ledger.wait_for_approval(
   714→                    action_id, timeout=timeout,
   715→                )
   716→        except asyncio.TimeoutError:
   717→            approved = False
   718→
   719→        status = "approved" if approved else (
   720→            "timeout" if not approved and ledger.pending_actions.get(action_id, {}).get("status") == "pending"
   721→            else "rejected"
   722→        )
   723→
   724→        return {
   725→            "status": status,
   726→            "action_id": action_id,
   727→            "component": component,
   728→            "action": action,
   729→            "message": (
   730→                f"Action {action_id} {status} by human"
   731→                if approved
   732→                else f"Action {action_id} not approved within {timeout}s"
   733→            ),
   734→            "receipt": self._make_receipt("request_approval", {
   735→                "status": status,
   736→                "receipt": {
   737→                    "action_id": action_id,
   738→                    "component": component,
   739→                    "action": action,
   740→                    "risk_level": risk_level,
   741→                    "description": description,
   742→                },
   743→            }),
   744→        }
   745→
   746→    # ─── Dispatch ───────────────────────────────────────────────────────
   747→
   748→    async def dispatch(self, tool_name: str, params: dict) -> dict:
   749→        """Dispatch a tool call to the appropriate handler.
   750→
   751→        Accepts both the bare tool name (e.g. 'chat') and the
   752→        namespaced form (e.g. 'friday.chat').
   753→        """
   754→        # Strip the 'friday.' prefix if present
   755→        if tool_name.startswith("friday."):
   756→            tool_name = tool_name[len("friday."):]
   757→
   758→        handlers = {
   759→            "chat": self.handle_chat,
   760→            "vision": self.handle_vision,
   761→            "search": self.handle_web_search,
   762→            "web_search": self.handle_web_search,
   763→            "generate_image": self.handle_image_generation,
   764→            "image_generation": self.handle_image_generation,
   765→            "generate_video": self.handle_video_generation,
   766→            "video_generation": self.handle_video_generation,
   767→            "code_execution": self.handle_code_execution,
   768→            "execute_action": self.handle_execute_action,
   769→            "request_approval": self.handle_request_approval,
   770→        }
   771→
   772→        handler = handlers.get(tool_name)
   773→        if not handler:
   774→            return {"status": "error", "message": f"Unknown tool: {tool_name}"}
   775→
   776→        return await handler(params)
   777→

   779→# ──────────────────────────────────────────────────────────────────────────────
   780→# JSON-RPC / stdio transport (MCP protocol)
   781→# ──────────────────────────────────────────────────────────────────────────────
   782→
   783→async def serve_stdio():
   784→    """Serve the MCP server over stdio using JSON-RPC."""
   785→    server = FridayMCPServer()
   786→
   787→    logger.info("FRIDAY MCP Server starting (stdio transport)")
   788→
   789→    reader = asyncio.StreamReader()
   790→    protocol = asyncio.StreamReaderProtocol(reader)
   791→    await asyncio.get_event_loop().connect_read_pipe(lambda: protocol, sys.stdin)
   792→
   793→    writer_transport, writer_protocol = await asyncio.get_event_loop().connect_write_pipe(
   794→        asyncio.streams.FlowControlMixin, sys.stdout
   795→    )
   796→    writer = asyncio.StreamWriter(writer_transport, writer_protocol, reader, asyncio.get_event_loop())
   797→
   798→    while True:
   799→        try:
   800→            line = await reader.readline()
   801→            if not line:
   802→                break
   803→
   804→            line = line.decode("utf-8").strip()
   805→            if not line:
   806→                continue
   807→
   808→            try:
   809→                request = json.loads(line)
   810→            except json.JSONDecodeError:
   811→                response = {"jsonrpc": "2.0", "error": {"code": -32700, "message": "Parse error"}, "id": None}
   812→                writer.write((json.dumps(response) + "\n").encode())
   813→                await writer.drain()
   814→                continue
   815→
   816→            request_id = request.get("id")
   817→            method = request.get("method", "")
   818→            params = request.get("params", {})
   819→
   820→            # ── JSON-RPC method routing ─────────────────────────────
   821→            if method == "initialize":
   822→                response = {
   823→                    "jsonrpc": "2.0",
   824→                    "id": request_id,
   825→                    "result": {
   826→                        "protocolVersion": "2024-11-05",
   827→                        "capabilities": {
   828→                            "tools": {"listChanged": False},
   829→                        },
   830→                        "serverInfo": {
   831→                            "name": "friday-mcp",
   832→                            "version": "1.0.0",
   833→                        },
   834→                    },
   835→                }
   836→            elif method == "tools/list":
   837→                response = {
   838→                    "jsonrpc": "2.0",
   839→                    "id": request_id,
   840→                    "result": {"tools": TOOLS},
   841→                }
   842→            elif method == "tools/call":
   843→                tool_name = params.get("name", "")
   844→                tool_params = params.get("arguments", {})
   845→                result = await server.dispatch(tool_name, tool_params)
   846→                response = {
   847→                    "jsonrpc": "2.0",
   848→                    "id": request_id,
   849→                    "result": {
   850→                        "content": [
   851→                            {"type": "text", "text": json.dumps(result, default=str)},
   852→                        ],
   853→                    },
   854→                }
   855→            elif method.startswith("friday."):
   856→                # Direct namespaced method call (e.g. friday.chat)
   857→                # Routes through the same dispatch as tools/call.
   858→                result = await server.dispatch(method, params)
   859→                response = {
   860→                    "jsonrpc": "2.0",
   861→                    "id": request_id,
   862→                    "result": {
   863→                        "content": [
   864→                            {"type": "text", "text": json.dumps(result, default=str)},
   865→                        ],
   866→                        "receipt": result.get("receipt"),
   867→                    },
   868→                }
   869→            elif method == "notifications/initialized":
   870→                # No response needed for notifications
   871→                continue
   872→            else:
   873→                response = {
   874→                    "jsonrpc": "2.0",
   875→                    "id": request_id,
   876→                    "error": {"code": -32601, "message": f"Method not found: {method}"},
   877→                }
   878→
   879→            writer.write((json.dumps(response, default=str) + "\n").encode())
   880→            await writer.drain()
   881→
   882→        except Exception as exc:
   883→            logger.error(f"MCP server error: {exc}")
   884→            try:
   885→                error_response = {
   886→                    "jsonrpc": "2.0",
   887→                    "error": {"code": -32603, "message": "Internal error"},
   888→                    "id": None,
   889→                }
   890→                writer.write((json.dumps(error_response) + "\n").encode())
   891→                await writer.drain()
   892→            except Exception as e:
   893→                logger.debug(f"Non-critical error: {e}")
   894→
   895→
   896→# ──────────────────────────────────────────────────────────────────────────────
   897→# Entry point
   898→# ──────────────────────────────────────────────────────────────────────────────
   899→
   900→if __name__ == "__main__":
   901→    asyncio.run(serve_stdio())
