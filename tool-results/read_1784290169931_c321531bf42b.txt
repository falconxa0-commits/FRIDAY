   187→   187→class FridayMCPServer:
   188→   188→    """FRIDAY MCP Server — wraps the action layer with ledger/auth/receipt."""
   189→   189→
   190→   190→    def __init__(self):
   191→   191→        self._glm_brain = None
   192→   192→        self._ledger = None
   193→   193→
   194→   194→    def _get_glm_brain(self):
   195→   195→        """Lazy-initialize GLMBrain."""
   196→   196→        if self._glm_brain is None:
   197→   197→            try:
   198→   198→                from core.glm_brain import GLMBrain
   199→   199→                self._glm_brain = GLMBrain()
   200→   200→            except Exception as exc:
   201→   201→                logger.error(f"Failed to initialize GLMBrain: {exc}")
   202→   202→        return self._glm_brain
   203→   203→
   204→   204→    def _get_ledger(self):
   205→   205→        """Lazy-initialize the action ledger."""
   206→   206→        if self._ledger is None:
   207→   207→            try:
   208→   208→                from core.ledger import get_ledger
   209→   209→                self._ledger = get_ledger()
   210→   210→            except Exception as exc:
   211→   211→                logger.error(f"Failed to initialize ledger: {exc}")
   212→   212→        return self._ledger
   213→   213→
   214→   214→    def _make_receipt(self, action: str, result: dict) -> dict:
   215→   215→        """Create a receipt for an MCP tool call."""
   216→   216→        return {
   217→   217→            "action": action,
   218→   218→            "timestamp": datetime.datetime.now().isoformat(),
   219→   219→            "status": result.get("status", "unknown"),
   220→   220→            "data": result.get("receipt", result.get("message", "")),
   221→   221→        }

   225→   225→    async def handle_chat(self, params: dict) -> dict:
   226→   226→        """Handle the 'chat' tool — logged in audit trail (read-only, no ledger gate needed)."""
   227→   227→        message = params.get("message", "")
   228→   228→        if not message:
   229→   229→            return {"status": "error", "message": "No message provided."}
   230→   230→
   231→   231→        # Log in audit trail (read-only calls don't need approval gate,
   232→   232→        # but must be logged for transparency)
   233→   233→        ledger = self._get_ledger()
   234→   234→        if ledger:
   235→   235→            try:
   236→   236→                action_id = ledger.queue_action(
   237→   237→                    "MCPChat", "chat", {"message_length": len(message)},
   238→   238→                    risk_level="low",
   239→   239→                )
   240→   240→                # Read-only calls auto-approve at STANDARD+ profile
   241→   241→                ledger.approve_action(action_id)
   242→   242→            except Exception as e:
   243→   243→                logger.debug(f"Non-critical error: {e}")
   244→   244→
   245→   245→        glm = self._get_glm_brain()
   246→   246→        if glm and glm.available():
   247→   247→            full_response = ""
   248→   248→            async for chunk in glm.chat_stream(message):
   249→   249→                full_response += chunk
   250→   250→            return {
   251→   251→                "status": "success",
   252→   252→                "message": full_response,
   253→   253→                "receipt": self._make_receipt("chat", {"status": "success", "receipt": {
   254→   254→                    "provider": "glm",
   255→   255→                    "response_length": len(full_response),
   256→   256→                }}),
   257→   257→            }
   258→   258→
   259→   259→        # Fallback to FridayBrain
   260→   260→        try:
   261→   261→            from core.brain import FridayBrain
   262→   262→            brain = FridayBrain()
   263→   263→            full_response = ""
   264→   264→            async for chunk in brain.chat_stream(message):
   265→   265→                full_response += chunk
   266→   266→            return {
   267→   267→                "status": "success",
   268→   268→                "message": full_response,
   269→   269→                "receipt": self._make_receipt("chat", {"status": "success", "receipt": {
   270→   270→                    "provider": brain.get_stats().get("provider", "unknown"),
   271→   271→                    "response_length": len(full_response),
   272→   272→                }}),
   273→   273→            }
   274→   274→        except Exception as exc:
   275→   275→            return {"status": "error", "message": str(exc)}

   346→   346→    async def handle_image_generation(self, params: dict) -> dict:
   347→   347→        """Handle the 'image_generation' tool — gated through ledger."""
   348→   348→        prompt = params.get("prompt", "")
   349→   349→        size = params.get("size", "1024x1024")
   350→   350→
   351→   351→        if not prompt:
   352→   352→            return {"status": "error", "message": "No prompt provided."}
   353→   353→
   354→   354→        # Gate through ledger — FAIL CLOSED if ledger unavailable
   355→   355→        ledger = self._get_ledger()
   356→   356→        if ledger is None:
   357→   357→            return {
   358→   358→                "status": "error",
   359→   359→                "message": "Ledger unavailable — refusing to execute without approval gate.",
   360→   360→                "receipt": self._make_receipt("image_generation", {"status": "error"}),
   361→   361→            }
   362→   362→        action_id = ledger.queue_action(
   363→   363→            "ImageGen", "generate_image",
   364→   364→            {"prompt": prompt, "size": size},
   365→   365→            risk_level="high",
   366→   366→        )
   367→   367→        # In MCP context, auto-approve is blocked by NEVER_AUTO_APPROVE_COMPONENTS
   368→   368→        # We need explicit approval — for now, reject if not approved
   369→   369→        approved = await ledger.wait_for_approval(action_id, timeout=30)
   370→   370→        if not approved:
   371→   371→            return {
   372→   372→                "status": "error",
   373→   373→                "message": "Image generation requires explicit approval. Action queued but not approved within timeout.",
   374→   374→                "receipt": self._make_receipt("image_generation", {"status": "rejected"}),
   375→   375→            }
   376→   376→
   377→   377→        try:
   378→   378→            from integrations.image_gen import ImageGen
   379→   379→            ig = ImageGen()
   380→   380→            result = await ig.execute("generate_image", {"prompt": prompt, "size": size})
   381→   381→            return {
   382→   382→                "status": result.get("status", "unknown"),
   383→   383→                "message": result.get("message", ""),
   384→   384→                "receipt": self._make_receipt("image_generation", result),
   385→   385→            }
   386→   386→        except Exception as exc:
   387→   387→            return {"status": "error", "message": str(exc)}

   470→   470→    async def handle_execute_action(self, params: dict) -> dict:
   471→   471→        """Handle 'friday.execute_action' — routes to UniversalConnector.
   472→   472→
   473→   473→        All actions go through the ledger's NEVER_AUTO_APPROVE_COMPONENTS
   474→   474→        gate. Physical/financial/code-execution actions require explicit
   475→   475→        user approval regardless of autonomy profile.
   476→   476→        """
   477→   477→        service = params.get("service") or params.get("component")
   478→   478→        action = params.get("action")
   479→   479→        action_params = params.get("params") or params.get("args") or {}
   480→   480→
   481→   481→        if not service or not action:
   482→   482→            return {
   483→   483→                "status": "error",
   484→   484→                "message": "Missing required parameters: service and action",
   485→   485→            }
   486→   486→
   487→   487→        try:
   488→   488→            from core.universal_connector import UniversalConnector
   489→   489→            connector = UniversalConnector()
   490→   490→            # Use a short approval timeout so MCP clients get a clear
   491→   491→            # "pending approval" response quickly rather than hanging.
   492→   492→            result = await connector.execute_action(
   493→   493→                service, action, action_params, approval_timeout=5,
   494→   494→            )
   495→   495→            return {
   496→   496→                "status": result.get("status", "unknown"),
   497→   497→                "message": result.get("message", ""),
   498→   498→                "receipt": self._make_receipt(
   499→   499→                    f"execute_action:{service}.{action}", result,
   500→   500→                ),
   501→   501→            }
   502→   502→        except Exception as exc:
   503→   503→            return {
   504→   504→                "status": "error",
   505→   505→                "message": f"execute_action failed: {exc}",
   506→   506→                "receipt": self._make_receipt("execute_action", {"status": "error"}),
   507→   507→            }
   508→   508→
   509→   509→    async def handle_request_approval(self, params: dict) -> dict:
   510→   510→        """Handle 'friday.request_approval' — external agent submits an action
   511→   511→        for human approval via Friday's ledger.
   512→   512→
   513→   513→        This is the killer feature for trusted execution: another AI agent
   514→   514→        (Claude Code, Cursor, etc.) can say "I want to delete this file"
   515→   515→        and Friday's human-in-the-loop gate stands between the suggestion
   516→   516→        and the action.
   517→   517→
   518→   518→        Friday speaks the request out loud (if voice is configured), waits
   519→   519→        for the human's real voice/keyboard response, and returns
   520→   520→        approved/rejected to the calling agent.
   521→   521→
   522→   522→        Params:
   523→   523→            component: The integration component (e.g. "Filesystem")
   524→   524→            action: The action name (e.g. "delete_file")
   525→   525→            params: Action parameters
   526→   526→            risk_level: "low" | "medium" | "high" | "critical"
   527→   527→            description: Human-readable description of what the agent wants to do
   528→   528→            timeout: Seconds to wait for approval (default 60)
   529→   529→            use_voice: If True, use voice-native approval (default False)
   530→   530→
   531→   531→        Returns:
   532→   532→            {status: "approved" | "rejected" | "timeout",
   533→   533→             action_id: str,
   534→   534→             receipt: dict}
   535→   535→        """
   536→   536→        component = params.get("component")
   537→   537→        action = params.get("action")
   538→   538→        action_params = params.get("params") or {}
   539→   539→        risk_level = params.get("risk_level", "high")
   540→   540→        description = params.get("description", f"{component}.{action}")
   541→   541→        timeout = int(params.get("timeout", 60))
   542→   542→        use_voice = bool(params.get("use_voice", False))
   543→   543→
   544→   544→        if not component or not action:
   545→   545→            return {
   546→   546→                "status": "error",
   547→   547→                "message": "Missing required parameters: component and action",
   548→   548→                "receipt": self._make_receipt("request_approval", {"status": "error"}),
   549→   549→            }
   550→   550→
   551→   551→        ledger = self._get_ledger()
   552→   552→        if not ledger:
   553→   553→            return {
   554→   554→                "status": "error",
   555→   555→                "message": "Ledger not available",
   556→   556→                "receipt": self._make_receipt("request_approval", {"status": "error"}),
   557→   557→            }
   558→   558→
   559→   559→        # Queue the action in the ledger (this is the gate)
   560→   560→        action_id = ledger.queue_action(
   561→   561→            component, action, action_params, risk_level=risk_level,
   562→   562→        )
   563→   563→        logger.info(
   564→   564→            "MCP request_approval: queued %s.%s (id=%s, risk=%s, timeout=%ss)",
   565→   565→            component, action, action_id, risk_level, timeout,
   566→   566→        )
   567→   567→
   568→   568→        # Wait for approval
   569→   569→        try:
   570→   570→            if use_voice:
   571→   571→                approved = await ledger.wait_for_voice_approval(
   572→   572→                    action_id, timeout=timeout,
   573→   573→                )
   574→   574→            else:
   575→   575→                approved = await ledger.wait_for_approval(
   576→   576→                    action_id, timeout=timeout,
   577→   577→                )
   578→   578→        except asyncio.TimeoutError:
   579→   579→            approved = False
   580→   580→
   581→   581→        status = "approved" if approved else (
   582→   582→            "timeout" if not approved and ledger.pending_actions.get(action_id, {}).get("status") == "pending"
   583→   583→            else "rejected"
   584→   584→        )
   585→   585→
   586→   586→        return {
   587→   587→            "status": status,
   588→   588→            "action_id": action_id,
   589→   589→            "component": component,
   590→   590→            "action": action,
   591→   591→            "message": (
   592→   592→                f"Action {action_id} {status} by human"
   593→   593→                if approved
   594→   594→                else f"Action {action_id} not approved within {timeout}s"
   595→   595→            ),
   596→   596→            "receipt": self._make_receipt("request_approval", {
   597→   597→                "status": status,
   598→   598→                "receipt": {
   599→   599→                    "action_id": action_id,
   600→   600→                    "component": component,
   601→   601→                    "action": action,
   602→   602→                    "risk_level": risk_level,
   603→   603→                    "description": description,
   604→   604→                },
   605→   605→            }),
   606→   606→        }
   607→   607→
   608→   608→    # ─── Dispatch ───────────────────────────────────────────────────────
   609→   609→
   610→   610→    async def dispatch(self, tool_name: str, params: dict) -> dict:
   611→   611→        """Dispatch a tool call to the appropriate handler.
   612→   612→
   613→   613→        Accepts both the bare tool name (e.g. 'chat') and the
   614→   614→        namespaced form (e.g. 'friday.chat').
   615→   615→        """
   616→   616→        # Strip the 'friday.' prefix if present
   617→   617→        if tool_name.startswith("friday."):
   618→   618→            tool_name = tool_name[len("friday."):]
   619→   619→
   620→   620→        handlers = {
   621→   621→            "chat": self.handle_chat,
   622→   622→            "vision": self.handle_vision,
   623→   623→            "search": self.handle_web_search,
   624→   624→            "web_search": self.handle_web_search,
   625→   625→            "generate_image": self.handle_image_generation,
   626→   626→            "image_generation": self.handle_image_generation,
   627→   627→            "generate_video": self.handle_video_generation,
   628→   628→            "video_generation": self.handle_video_generation,
   629→   629→            "code_execution": self.handle_code_execution,
   630→   630→            "execute_action": self.handle_execute_action,
   631→   631→            "request_approval": self.handle_request_approval,
   632→   632→        }
   633→   633→
   634→   634→        handler = handlers.get(tool_name)
   635→   635→        if not handler:
   636→   636→            return {"status": "error", "message": f"Unknown tool: {tool_name}"}
   637→   637→
   638→   638→        return await handler(params)
   639→   639→

   641→   641→# ──────────────────────────────────────────────────────────────────────────────
   642→   642→# JSON-RPC / stdio transport (MCP protocol)
   643→   643→# ──────────────────────────────────────────────────────────────────────────────
   644→   644→
   645→   645→async def serve_stdio():
   646→   646→    """Serve the MCP server over stdio using JSON-RPC."""
   647→   647→    server = FridayMCPServer()
   648→   648→
   649→   649→    logger.info("FRIDAY MCP Server starting (stdio transport)")
   650→   650→
   651→   651→    reader = asyncio.StreamReader()
   652→   652→    protocol = asyncio.StreamReaderProtocol(reader)
   653→   653→    await asyncio.get_event_loop().connect_read_pipe(lambda: protocol, sys.stdin)
   654→   654→
   655→   655→    writer_transport, writer_protocol = await asyncio.get_event_loop().connect_write_pipe(
   656→   656→        asyncio.streams.FlowControlMixin, sys.stdout
   657→   657→    )
   658→   658→    writer = asyncio.StreamWriter(writer_transport, writer_protocol, reader, asyncio.get_event_loop())
   659→   659→
   660→   660→    while True:
   661→   661→        try:
   662→   662→            line = await reader.readline()
   663→   663→            if not line:
   664→   664→                break
   665→   665→
   666→   666→            line = line.decode("utf-8").strip()
   667→   667→            if not line:
   668→   668→                continue
   669→   669→
   670→   670→            try:
   671→   671→                request = json.loads(line)
   672→   672→            except json.JSONDecodeError:
   673→   673→                response = {"jsonrpc": "2.0", "error": {"code": -32700, "message": "Parse error"}, "id": None}
   674→   674→                writer.write((json.dumps(response) + "\n").encode())
   675→   675→                await writer.drain()
   676→   676→                continue
   677→   677→
   678→   678→            request_id = request.get("id")
   679→   679→            method = request.get("method", "")
   680→   680→            params = request.get("params", {})
   681→   681→
   682→   682→            # ── JSON-RPC method routing ─────────────────────────────
   683→   683→            if method == "initialize":
   684→   684→                response = {
   685→   685→                    "jsonrpc": "2.0",
   686→   686→                    "id": request_id,
   687→   687→                    "result": {
   688→   688→                        "protocolVersion": "2024-11-05",
   689→   689→                        "capabilities": {
   690→   690→                            "tools": {"listChanged": False},
   691→   691→                        },
   692→   692→                        "serverInfo": {
   693→   693→                            "name": "friday-mcp",
   694→   694→                            "version": "1.0.0",
   695→   695→                        },
   696→   696→                    },
   697→   697→                }
   698→   698→            elif method == "tools/list":
   699→   699→                response = {
   700→   700→                    "jsonrpc": "2.0",
   701→   701→                    "id": request_id,
   702→   702→                    "result": {"tools": TOOLS},
   703→   703→                }
   704→   704→            elif method == "tools/call":
   705→   705→                tool_name = params.get("name", "")
   706→   706→                tool_params = params.get("arguments", {})
   707→   707→                result = await server.dispatch(tool_name, tool_params)
   708→   708→                response = {
   709→   709→                    "jsonrpc": "2.0",
   710→   710→                    "id": request_id,
   711→   711→                    "result": {
   712→   712→                        "content": [
   713→   713→                            {"type": "text", "text": json.dumps(result, default=str)},
   714→   714→                        ],
   715→   715→                    },
   716→   716→                }
   717→   717→            elif method.startswith("friday."):
   718→   718→                # Direct namespaced method call (e.g. friday.chat)
   719→   719→                # Routes through the same dispatch as tools/call.
   720→   720→                result = await server.dispatch(method, params)
   721→   721→                response = {
   722→   722→                    "jsonrpc": "2.0",
   723→   723→                    "id": request_id,
   724→   724→                    "result": {
   725→   725→                        "content": [
   726→   726→                            {"type": "text", "text": json.dumps(result, default=str)},
   727→   727→                        ],
   728→   728→                        "receipt": result.get("receipt"),
   729→   729→                    },
   730→   730→                }
   731→   731→            elif method == "notifications/initialized":
   732→   732→                # No response needed for notifications
   733→   733→                continue
   734→   734→            else:
   735→   735→                response = {
   736→   736→                    "jsonrpc": "2.0",
   737→   737→                    "id": request_id,
   738→   738→                    "error": {"code": -32601, "message": f"Method not found: {method}"},
   739→   739→                }
   740→   740→
   741→   741→            writer.write((json.dumps(response, default=str) + "\n").encode())
   742→   742→            await writer.drain()
   743→   743→
   744→   744→        except Exception as exc:
   745→   745→            logger.error(f"MCP server error: {exc}")
   746→   746→            try:
   747→   747→                error_response = {
   748→   748→                    "jsonrpc": "2.0",
   749→   749→                    "error": {"code": -32603, "message": "Internal error"},
   750→   750→                    "id": None,
   751→   751→                }
   752→   752→                writer.write((json.dumps(error_response) + "\n").encode())
   753→   753→                await writer.drain()
   754→   754→            except Exception as e:
   755→   755→                logger.debug(f"Non-critical error: {e}")
   756→   756→
   757→   757→
   758→   758→# ──────────────────────────────────────────────────────────────────────────────
   759→   759→# Entry point
   760→   760→# ──────────────────────────────────────────────────────────────────────────────
   761→   761→
   762→   762→if __name__ == "__main__":
   763→   763→    asyncio.run(serve_stdio())
