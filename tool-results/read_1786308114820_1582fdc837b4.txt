   489→
   490→def _plugin_install(name: str) -> int:
   491→    """Install a plugin from the marketplace into integrations/.
   492→
   493→    Security: Before copying, the plugin source is statically scanned
   494→    for dangerous top-level imports (os, subprocess, socket, shlex,
   495→    ctypes, etc.). Plugins that import these at module level are
   496→    refused — the user must explicitly review and copy them manually.
   497→    This prevents trivial supply-chain attacks via the marketplace.
   498→    """
   499→    from pathlib import Path
   500→    import shutil
   501→    import ast
   502→
   503→    src_dir = Path(__file__).resolve().parent.parent / "marketplace" / "plugins" / name
   504→    if not src_dir.is_dir():
   505→        return _err(f"Plugin '{name}' not found in marketplace/")
   506→    # Find the plugin .py file inside the directory
   507→    plugin_files = list(src_dir.glob("*.py"))
   508→    if not plugin_files:
   509→        return _err(f"Plugin '{name}' has no .py file")
   510→    target = Path(__file__).resolve().parent.parent / "integrations" / plugin_files[0].name
   511→    if target.exists():
   512→        console.print(f"[dim]{name} already installed.[/]")
   513→        return 0
   514→
   515→    # ---- Security scan: refuse dangerous imports anywhere in the AST ----
   516→    # These imports grant the plugin full process privileges (file access,
   517→    # subprocess, network, foreign function interface). They are refused
   518→    # regardless of where they appear in the source — top-level, inside
   519→    # functions, inside classes, inside conditionals, inside try/except —
   520→    # because a plugin can call any of them at runtime.
   521→    #
   522→    # NOTE: This is a defense-in-depth measure, not a sandbox. A
   523→    # determined attacker can still bypass it (e.g., via __import__,
   524→    # importlib.import_module with a computed name, or by reading the
   525→    # module file at runtime and exec'ing it). For true isolation, run
   526→    # plugins in a subprocess with seccomp/bubblewrap. This scan catches
   527→    # the most common supply-chain patterns: top-level or lazy imports
   528→    # of os/subprocess/socket/etc.
   529→    DANGEROUS_IMPORTS = {
   530→        "os", "subprocess", "socket", "shlex", "ctypes", "sys",
   531→        "importlib", "builtins", "pty", "multiprocessing",
   532→    }
   533→    DANGEROUS_BUILTINS = {
   534→        "__import__", "exec", "eval", "compile",
   535→    }
   536→    src_text = plugin_files[0].read_text()
   537→    try:
   538→        tree = ast.parse(src_text)
   539→    except SyntaxError as exc:
   540→        return _err(f"Plugin '{name}' has a syntax error: {exc}")
   541→
   542→    found_dangerous = []
   543→
   544→    # Walk the ENTIRE AST — not just top-level nodes. This catches
   545→    # lazy imports inside __init__, methods, conditionals, loops, etc.
   546→    # (The previous implementation only walked ast.iter_child_nodes(tree),
   547→    # which missed imports inside any nested scope.)
   548→    for node in ast.walk(tree):
   549→        # Direct `import os` / `import subprocess as sp` (anywhere)
   550→        if isinstance(node, ast.Import):
   551→            for alias in node.names:
   552→                root = alias.name.split(".")[0]
   553→                if root in DANGEROUS_IMPORTS:
   554→                    found_dangerous.append(
   555→                        f"import {alias.name}  (line {node.lineno})"
   556→                    )
   557→        # `from os import ...` / `from subprocess import ...` (anywhere)
   558→        elif isinstance(node, ast.ImportFrom):
   559→            if node.module:
   560→                root = node.module.split(".")[0]
   561→                if root in DANGEROUS_IMPORTS:
   562→                    found_dangerous.append(
   563→                        f"from {node.module} import ...  (line {node.lineno})"
   564→                    )
   565→        # Catch __import__("subprocess"), exec(...), eval(...), compile(...)
   566→        # used as a dynamic-import escape hatch.
   567→        elif isinstance(node, ast.Call):
   568→            func = node.func
   569→            if isinstance(func, ast.Name) and func.id in DANGEROUS_BUILTINS:
   570→                found_dangerous.append(
   571→                    f"{func.id}(...)  (line {node.lineno}) — "
   572→                    "dynamic code execution"
   573→                )
   574→            elif isinstance(func, ast.Attribute) and func.attr == "__import__":
   575→                found_dangerous.append(
   576→                    f"__import__(...)  (line {node.lineno}) — "
   577→                    "dynamic import escape hatch"
   578→                )
   579→
   580→    if found_dangerous:
   581→        console.print(
   582→            Panel(

   657→# Subcommand map: name → (handler, takes_args)
   658→# Flag-style entries start with --
   659→COMMANDS = {
   660→    "chat":       cmd_chat,
   661→    "research":   cmd_research,
   662→    "generate":   cmd_generate,
   663→    "video":      cmd_video,
   664→    "morning":    cmd_morning,
   665→    "council":    cmd_council,
   666→    "status":     cmd_status,
   667→    "memory":     cmd_memory,
   668→    "bench":      cmd_bench,
   669→    "trust":      cmd_trust,
   670→    "ledger":     cmd_ledger,
   671→    "plugin":     cmd_plugin,
   672→    "--watch":    cmd_watch,
   673→    "--voice":    cmd_voice,
   674→    "--council":  cmd_council_tui,
   675→}
   676→
   677→
   678→def dispatch(argv: list[str]) -> int:
   679→    if not argv:
   680→        # Default: TUI
   681→        return asyncio.run(_run_tui_default())
   682→
   683→    cmd = argv[0]
   684→    rest = argv[1:]
   685→
   686→    if cmd in ("-h", "--help", "help"):
   687→        _print_help()
   688→        return 0
   689→
   690→    handler = COMMANDS.get(cmd)
   691→    if handler is None:
   692→        return _err(f"Unknown command: {cmd}\nRun 'friday help' for usage.")
   693→    return asyncio.run(handler(rest))
   694→
   695→
   696→async def _run_tui_default() -> int:
   697→    """Boot the TUI with the default provider.
   698→
   699→    Handles KeyboardInterrupt gracefully — Ctrl+C exits the TUI
   700→    without printing a traceback.
   701→    """
   702→    status = await boot_sequence()
   703→    from cli.terminal import run_tui
   704→    try:
   705→        await run_tui(status)
   706→    except KeyboardInterrupt:
   707→        # Graceful exit — no traceback, no undefined-logger crash.
   708→        # The previous implementation referenced an undefined `logger`
   709→        # symbol here, which itself raised NameError on Ctrl+C.
   710→        console.print("\n[dim]Goodbye.[/]")
   711→    return 0
